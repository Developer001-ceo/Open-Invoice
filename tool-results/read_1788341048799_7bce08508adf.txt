    19→// ── Shift+S Keyboard Hint (Recommendation 4) ──────────────────────────────────
    20→// Shows a subtle hint near the bottom of the canvas when an element is selected.
    21→// Fades away permanently after the user uses Shift+S a few times.
    22→const MAX_HINT_DISPLAYS = 3; // Show hint for the first 3 element selections
    23→
    24→function ShiftSHint() {
    25→  const selectedElementIds = useDesignerStore((s) => s.selectedElementIds);
    26→  const spotlightHintShown = useDesignerStore((s) => s.spotlightHintShown);
    27→  const setSpotlightHintShown = useDesignerStore((s) => s.setSpotlightHintShown);
    28→  const spotlightOpen = useDesignerStore((s) => s.spotlightOpen);
    29→
    30→  const [visible, setVisible] = useState(false);
    31→  const [dismissed, setDismissed] = useState(false);
    32→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    33→  const hideTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    34→
    35→  const hasSelection = selectedElementIds.length > 0;
    36→  // Derive reset state: when no selection or spotlight is open, treat as dismissed
    37→  const isActive = hasSelection && !spotlightOpen;
    38→  // Dismissed only applies while active; effectively resets when selection is cleared
    39→  const effectiveDismissed = isActive && dismissed;
    40→  const shouldShow = isActive && spotlightHintShown < MAX_HINT_DISPLAYS && !effectiveDismissed;
    41→
    42→  useEffect(() => {
    43→    // Show hint when element is selected (if not already shown too many times)
    44→    if (shouldShow) {
    45→      // Small delay so it doesn't flash during quick selections
    46→      timerRef.current = setTimeout(() => {
    47→        setVisible(true);
    48→        setSpotlightHintShown(spotlightHintShown + 1);
    49→      }, 800);
    50→
    51→      // Auto-hide after 4 seconds
    52→      hideTimerRef.current = setTimeout(() => {
    53→        setVisible(false);
    54→        setDismissed(true);
    55→      }, 4800);
    56→
    57→      return () => {
    58→        if (timerRef.current) clearTimeout(timerRef.current);
    59→        if (hideTimerRef.current) clearTimeout(hideTimerRef.current);
    60→      };
    61→    }
    62→  }, [shouldShow, spotlightHintShown, setSpotlightHintShown]);
    63→
    64→  if (!visible || spotlightOpen) return null;
    65→
    66→  return (
    67→    <div
    68→      className="absolute bottom-4 left-1/2 -translate-x-1/2 z-[9999] pointer-events-none"
    69→      style={{
    70→        opacity: visible ? 1 : 0,
    71→        transform: `translateX(-50%) translateY(${visible ? 0 : 8}px)`,
    72→        transition: 'opacity 400ms ease-out, transform 400ms ease-out',
    73→      }}
    74→    >
    75→      <div className="bg-card/90 backdrop-blur-sm border border-border/60 rounded-lg px-4 py-2 shadow-lg flex items-center gap-2">
    76→        <span className="text-xs text-muted-foreground">Press</span>
    77→        <kbd className="px-2 py-0.5 rounded bg-muted border border-border font-mono text-[11px] text-foreground font-medium">
    78→          Shift+S
    79→        </kbd>
    80→        <span className="text-xs text-muted-foreground">for quick properties</span>
    81→      </div>
    82→    </div>
    83→  );
    84→}

   153→  // ── Cursor lock during active resize / corner-drag ─────────────────────────
   154→  // While the user is dragging a resize handle (isResizing) or an individual
   155→  // corner point (isCornerDragging), lock the cursor to the matching resize

   161→  const activeResizeCursor = useMemo(() => {

   174→      return getResizeCursor(handlePos, rotation);
   175→    }
   176→    return null;

   179→  // Apply the locked cursor globally so it persists everywhere the mouse goes
   180→  // during the drag — including off the handle, off the element body, off the
   181→  // canvas viewport (over panels/menus), and even off the browser window. A
   182→  // dynamically-injected <style> rule with `* { cursor: ... !important }` is
   183→  // used so it overrides EVERY competing cursor rule (handle inline styles,
   184→  // element `cursor-move` classes, panel cursors, etc.) — this is what
   185→  // eliminates the flicker. The rule is removed on drag end so normal
   186→  // per-element cursors resume.
   187→  useEffect(() => {
   188→    if (activeResizeCursor) {
   189→      const styleId = 'active-resize-cursor-lock';
   190→      let styleEl = document.getElementById(styleId) as HTMLStyleElement | null;
   191→      if (!styleEl) {
   192→        styleEl = document.createElement('style');
   193→        styleEl.id = styleId;
   194→        document.head.appendChild(styleEl);
   195→      }
   196→      // `*` selector + !important overrides all element-level cursor rules.
   197→      // We also set body directly as a belt-and-suspenders for browsers that
   198→      // ignore the `*` rule on the root element.
   199→      styleEl.textContent = `* { cursor: ${activeResizeCursor} !important; }`;
   200→      document.body.style.cursor = activeResizeCursor;
   201→      return () => {
   202→        if (styleEl.parentNode) styleEl.parentNode.removeChild(styleEl);
   203→        document.body.style.cursor = '';
   204→      };
   205→    }
   206→  }, [activeResizeCursor]);

   209→  // mouse position. When released, stop space-panning. This allows the user

   211→  // required.
   212→  // NOTE: The keydown/keyup handlers now start/stop panning SYNCHRONOUSLY

   240→      }
   241→    }
   242→  }, [spacePressed]);

   244→  // Center/re-center the page when dimensions change
   245→  const prevSizeRef = useRef({ w: pageWidth, h: pageHeight });
   246→  // Ref to hold RAF cleanup functions for the fit-on-mount effect
   247→  const rafRefCleanup = useRef<(() => void) | null>(null);
   248→
   249→  useEffect(() => {
   250→    const viewport = viewportRef.current;
   251→    if (!viewport) return;
   252→

   254→      prevSizeRef.current.w !== pageWidth ||
   255→      prevSizeRef.current.h !== pageHeight;

   258→

   260→    // the fit. On initial mount (and especially on reload), the properties panel
   261→    // animates its width from 0 → 288px, so a synchronous clientWidth read gives
   262→    // the wrong value. Waiting two frames ensures the layout has settled.
   263→    const fit = () => {
   264→      const vp = viewportRef.current;
   265→      if (!vp) return;

   267→      prevSizeRef.current = { w: pageWidth, h: pageHeight };
   268→      isReadyRef.current = true;
   269→    };

   271→      const raf2 = requestAnimationFrame(fit);
   272→      rafRefCleanup.current = () => cancelAnimationFrame(raf2);
   273→    });
   274→    rafRefCleanup.current = () => cancelAnimationFrame(raf1);
   275→
   276→    return () => {
   277→      if (rafRefCleanup.current) rafRefCleanup.current();
   278→    };
   279→  }, [pageWidth, pageHeight]);
   280→
   281→  // ResizeObserver: re-fit the page when the viewport's actual size changes
   282→  // during the initial mount settling period (e.g. when the properties panel
   283→  // finishes its width animation). After the settling period, viewport resizes
   284→  // (like window resize) don't override the user's zoom/pan.
   285→  const hasObservedResizeRef = useRef(false);
   286→  const settlingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   287→  const isSettlingRef = useRef(true);
   288→  useEffect(() => {
   289→    const viewport = viewportRef.current;
   290→    if (!viewport) return;
   291→    // After 1.5s, stop re-fitting on viewport resize (user is in control now)
   292→    settlingTimerRef.current = setTimeout(() => { isSettlingRef.current = false; }, 1500);
   293→    const observer = new ResizeObserver(() => {
   294→      if (!isSettlingRef.current) return;
   295→      // Skip the very first observation (it fires on mount with the initial size)
   296→      if (!hasObservedResizeRef.current) {
   297→        hasObservedResizeRef.current = true;
   298→        return;
   299→      }
   300→      const st = useDesignerStore.getState();
   301→      const vp = viewportRef.current;
   302→      if (!vp) return;
   303→      st.fitToPage(vp.clientWidth, vp.clientHeight);
   304→    });
   305→    observer.observe(viewport);
   306→    return () => {
   307→      observer.disconnect();
   308→      if (settlingTimerRef.current) clearTimeout(settlingTimerRef.current);
   309→    };
   310→  }, []);

   314→  const panCoordsRef = useRef<{ x: number; y: number } | null>(null);

   318→    const coords = panCoordsRef.current;

   320→    panCoordsRef.current = null;

   333→    const raf = selRafRef.current;
   334→    const coords = selCoordsRef.current;
   335→    selRafRef.current = null;
   336→    selCoordsRef.current = null;

   344→  // React re-render from blocking the main thread during fast mouse movement.
   345→  // Only the latest mouse position is processed per animation frame.
   346→  const cornerRafRef = useRef<number | null>(null);
   347→  const cornerCoordsRef = useRef<{ x: number; y: number } | null>(null);
   348→  const flushCornerRef = useRef(() => {});
   349→  flushCornerRef.current = () => {
   350→    const raf = cornerRafRef.current;
   351→    const coords = cornerCoordsRef.current;
   352→    cornerRafRef.current = null;
   353→    cornerCoordsRef.current = null;
   354→    if (raf) cancelAnimationFrame(raf);
   355→    if (coords) {
   356→      const s = useDesignerStore.getState();
   357→      if (s.isCornerDragging) {
   358→        s.updateCornerDrag(coords.x, coords.y, s.shiftHeld, s.ctrlHeld);
   359→      }
   360→    }
   361→  };
   362→
   363→  // Handle global mouse events
   364→  // IMPORTANT: We use getState() inside handlers for zoom/panX/panY to avoid
   365→  // re-registering listeners on every pan/zoom change, which caused an infinite
   366→  // re-render cycle (pan changes → effect re-registers → next mousemove → pan changes → ...)
   367→  useEffect(() => {
   368→    const handleMouseMove = (e: MouseEvent) => {
   369→      // Read latest state from store to avoid stale closures
   370→      const state = useDesignerStore.getState();
   371→
   372→      // When actively selecting, we MUST process ALL mousemove events so we can
   373→      // clamp the cursor to the viewport boundary — even when the cursor moves
   374→      // over panels during fast drags.  Skip the panel/input guards for selection.
   375→      // The same applies to any in-progress canvas operation (corner drag, move,
   376→      // resize, endpoint drag, table resize, group-child drag/resize): once the
   377→      // user has started the gesture on the canvas, the cursor will often sweep
   378→      // across the properties/left panel at high speed — if we `return` here the
   379→      // operation freezes mid-drag and only resumes when the cursor re-enters
   380→      // the canvas, which is the "stuck until I release" symptom. Bypass the
   381→      // panel/input guards for the duration of any active gesture.
   382→      const hasActiveGesture =
   383→        state.isSelecting ||
   384→        state.isCornerDragging ||
   385→        state.isDragging ||
   386→        state.isResizing ||
   387→        state.isEndpointDragging ||
   388→        state.isGroupChildDragging ||
   389→        state.isGroupChildResizing ||
   390→        state.isTableColResizing ||
   391→        state.isTableRowResizing ||
   392→        state.isPanning;
   393→      if (!hasActiveGesture) {
   394→        // Skip if user is interacting with a UI control outside the canvas
   395→        // (e.g., sliders, inputs, color pickers in the properties panel)
   396→        const target = e.target as HTMLElement;
   397→        if (target?.closest?.('[data-ui-panel]')) return;
   398→        if (target?.tagName === 'INPUT' || target?.tagName === 'SELECT' || target?.tagName === 'TEXTAREA') return;
   399→      }
   400→
   401→      // Space-panning: if spacebar is held but panning hasn't started yet
   402→      // (e.g., mouse wasn't over canvas when space was pressed), start now.
   403→      if (spacePanningRef.current && !state.isPanning) {
   404→        state.startPanning(e.clientX, e.clientY);
   405→      }
   406→
   407→      if (state.isPanning) {
   408→        // Throttle updatePan with RAF to avoid excessive state updates during panning
   409→        panCoordsRef.current = { x: e.clientX, y: e.clientY };
   410→        if (!panRafRef.current) {
   411→          panRafRef.current = requestAnimationFrame(() => flushPanRef.current());
   412→        }
   413→      }
   414→      if (state.isDragging) {
   415→        state.updateDrag(e.clientX, e.clientY);
   416→      }
   417→      if (state.isGroupChildDragging) {
   418→        state.updateGroupChildDrag(e.clientX, e.clientY);
   419→      }
   420→      if (state.isGroupChildResizing) {
   421→        state.updateGroupChildResize(e.clientX, e.clientY);
   422→      }
   423→      if (state.isResizing) {
   424→        state.updateResize(e.clientX, e.clientY);
   425→      }
   426→      if (state.isTableColResizing || state.isTableRowResizing) {
   427→        state.updateTableResize(e.clientX, e.clientY);
   428→        return;
   429→      }
   430→      if (state.isEndpointDragging) {
   431→        lastMouseRef.current = { x: e.clientX, y: e.clientY };
   432→        state.updateEndpointDrag(e.clientX, e.clientY, e.shiftKey);
   433→      }
   434→      if (state.isCornerDragging) {
   435→        lastMouseRef.current = { x: e.clientX, y: e.clientY };
   436→        // RAF-throttle: store the latest mouse position and process at most
   437→        // once per animation frame. This prevents fast mouse movement from
   438→        // queuing up stale store updates that cause visual lag.
   439→        cornerCoordsRef.current = { x: e.clientX, y: e.clientY };
   440→        if (!cornerRafRef.current) {
   441→          cornerRafRef.current = requestAnimationFrame(() => flushCornerRef.current());
   442→        }
   443→      }
   444→      if (state.isSelecting) {
   445→        // Clamp cursor to viewport boundaries so the selection rectangle
   446→        // never leaves the workspace area (between panels/menus).
   447→        const viewport = viewportRef.current;
   448→        if (!viewport) return;
   449→        const rect = viewport.getBoundingClientRect();
   450→        const clampedX = Math.max(rect.left, Math.min(e.clientX, rect.right));
   451→        const clampedY = Math.max(rect.top, Math.min(e.clientY, rect.bottom));
   452→        const canvasX = (clampedX - rect.left) / state.zoom - state.panX;
   453→        const canvasY = (clampedY - rect.top) / state.zoom - state.panY;
   454→        // RAF-throttle selection updates for smooth performance
   455→        selCoordsRef.current = { canvasX, canvasY };
