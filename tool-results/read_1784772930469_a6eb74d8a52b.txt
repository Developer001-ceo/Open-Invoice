   262→// ── Masonry Card Layout Calculation ────────────────────────────────────────
   263→// Places cards in a masonry-style layout with a fixed 20px gap between
   264→// all cards (both horizontally and vertically). Cards in the same column
   265→// are stacked vertically; each card goes into the shortest column.
   266→// Each card is horizontally centered within its column slot so that
   267→// narrower cards (260px, 280px) are visually centered rather than
   268→// left-aligned within the 320px column width.
   269→// Utilizes as much screen space as possible, ensures no card goes off-screen.
   270→
   271→const COLUMN_WIDTH = 320; // Column slot width (widest card) for masonry layout
   272→const CARD_GAP = 20; // Fixed 20px gap between all cards
   273→const EDGE_MARGIN = 20; // minimum distance from viewport edges
   274→
   275→interface CardPosition {
   276→  x: number;
   277→  y: number;
   278→}
   279→
   280→function getCardHeight(sectionId: string, measuredHeights?: Map<string, number>): number {
   281→  if (measuredHeights && measuredHeights.has(sectionId)) {
   282→    return measuredHeights.get(sectionId)!;
   283→  }
   284→  return estimateCardHeight(sectionId);
   285→}
   286→
   287→function calculateCardPositions(
   288→  sectionIds: string[],
   289→  containerWidth: number,
   290→  containerHeight: number,
   291→  measuredHeights?: Map<string, number>,
   292→): { positions: CardPosition[] } {
   293→  const count = sectionIds.length;
   294→  if (count === 0) return { positions: [] };
   295→
   296→  const availableWidth = containerWidth - 2 * EDGE_MARGIN;
   297→  const availableHeight = containerHeight - 2 * EDGE_MARGIN;
   298→
   299→  // Determine maximum number of columns that fit
   300→  const maxCols = Math.floor((availableWidth + CARD_GAP) / (COLUMN_WIDTH + CARD_GAP));
   301→
   302→  // Try to find the optimal number of columns that fits all cards on screen
   303→  // Start with fewer columns (for readability), increase if cards overflow vertically
   304→  let bestLayout: { positions: CardPosition[]; totalHeight: number } | null = null;
   305→
   306→  for (let tryCols = Math.max(1, Math.min(3, maxCols, count)); tryCols <= maxCols && tryCols <= count; tryCols++) {
   307→    const totalGridWidth = tryCols * COLUMN_WIDTH + (tryCols - 1) * CARD_GAP;
   308→    const startX = EDGE_MARGIN + (availableWidth - totalGridWidth) / 2;
   309→
   310→    const columnY: number[] = Array(tryCols).fill(0);
   311→    const positions: CardPosition[] = [];
   312→
   313→    for (let i = 0; i < count; i++) {
   314→      let shortestCol = 0;
   315→      let shortestY = columnY[0];
   316→      for (let c = 1; c < tryCols; c++) {
   317→        if (columnY[c] < shortestY) {
   318→          shortestY = columnY[c];
   319→          shortestCol = c;
   320→        }
   321→      }
   322→
   323→      // Center the card horizontally within its column slot
   324→      const cardActualWidth = getCardActualWidthPx(sectionIds[i]);
   325→      const centerOffset = (COLUMN_WIDTH - cardActualWidth) / 2;
   326→      const x = startX + shortestCol * (COLUMN_WIDTH + CARD_GAP) + centerOffset;
   327→      const y = columnY[shortestCol];
   328→
   329→      positions.push({ x, y });
   330→
   331→      const cardHeight = getCardHeight(sectionIds[i], measuredHeights);
   332→      columnY[shortestCol] = y + cardHeight + CARD_GAP;
   333→    }
   334→
   335→    // Calculate the total height of the grid (tallest column)
   336→    const totalHeight = Math.max(...columnY) - CARD_GAP; // subtract trailing gap
   337→
   338→    // If this layout fits on screen, use it
   339→    if (totalHeight <= availableHeight) {
   340→      // Vertically center the grid within the available height
   341→      const yOffset = EDGE_MARGIN + (availableHeight - totalHeight) / 2;
   342→      const centeredPositions = positions.map(p => ({ x: p.x, y: p.y + yOffset }));
   343→      bestLayout = { positions: centeredPositions, totalHeight };
   344→      break; // Found a fitting layout with minimal columns
   345→    }
   346→
   347→    // Track the best layout even if it overflows (in case nothing fits)
   348→    if (!bestLayout || totalHeight < bestLayout.totalHeight) {
   349→      const yOffset = EDGE_MARGIN;
   350→      const adjustedPositions = positions.map(p => ({ x: p.x, y: p.y + yOffset }));
   351→      bestLayout = { positions: adjustedPositions, totalHeight };
   352→    }
   353→  }
   354→
   355→  if (!bestLayout) {
   356→    // Fallback: just use 1 column, cards centered within column slot
   357→    const startX = EDGE_MARGIN;
   358→    const columnY: number[] = [0];
   359→    const positions: CardPosition[] = [];
   360→    for (let i = 0; i < count; i++) {
   361→      const cardActualWidth = getCardActualWidthPx(sectionIds[i]);
   362→      const centerOffset = (COLUMN_WIDTH - cardActualWidth) / 2;
   363→      positions.push({ x: startX + centerOffset, y: columnY[0] + EDGE_MARGIN });
   364→      columnY[0] += getCardHeight(sectionIds[i], measuredHeights) + CARD_GAP;
   365→    }
   366→    bestLayout = { positions, totalHeight: columnY[0] };
   367→  }
   368→
   369→  return { positions: bestLayout.positions };
   370→}

   385→export function SpotlightOverlay({ element, multiSelectIds }: { element: CanvasElement; multiSelectIds?: string[] }) {
   386→  const spotlightOpen = useDesignerStore((s) => s.spotlightOpen);
   387→  const closeSpotlight = useDesignerStore((s) => s.closeSpotlight);
   388→  const addFloatingCard = useDesignerStore((s) => s.addFloatingCard);
   389→  const floatingCards = useDesignerStore((s) => s.floatingCards);
   390→  const propertyDisplayMode = useDesignerStore((s) => s.propertyDisplayMode);
   391→
   392→  // Animation states
   393→  const [isVisible, setIsVisible] = useState(false);
   394→  const [cardsVisible, setCardsVisible] = useState(false);
   395→  const [hoveredCard, setHoveredCard] = useState<string | null>(null);
   396→  const [selectedCardId, setSelectedCardId] = useState<string | null>(null);
   397→  const [isDismissing, setIsDismissing] = useState(false);
   398→
   399→  // Measured card heights — two-pass render to avoid overlap from inaccurate estimates
   400→  const [measuredHeights, setMeasuredHeights] = useState<Map<string, number>>(new Map());
   401→
   402→  // Timer refs for cleanup
   403→  const cardTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   404→  const dismissTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   405→  const selectTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   406→  const rafRef = useRef<number | null>(null);
   407→
   408→  const isMultiSelect = !!(multiSelectIds && multiSelectIds.length > 1);
   409→  const sections = isMultiSelect ? getMultiSelectionSections() : getPropertySections(element.type);
   410→
   411→  // ── Derived layout (computed, not state) ────────────────────────────────
   412→
   413→  const dimensions = useMemo(() =>
   414→    spotlightOpen ? { width: window.innerWidth, height: window.innerHeight } : { width: 0, height: 0 },
   415→    [spotlightOpen],
   416→  );
   417→
   418→  // Memoize card layout calculation — uses measured heights when available
   419→  const sectionIds = useMemo(() => sections.map(s => s.id), [sections]);
   420→  const cardLayout = useMemo(() => {
   421→    if (!spotlightOpen || dimensions.width === 0) return { positions: [] as CardPosition[] };
   422→    return calculateCardPositions(sectionIds, dimensions.width, dimensions.height, measuredHeights);
   423→  }, [spotlightOpen, sectionIds, dimensions.width, dimensions.height, measuredHeights]);
   424→
   425→  // ── Open / Close lifecycle ──────────────────────────────────────────────
   426→
   427→  useEffect(() => {
   428→    // Cancel any pending rAF from a previous run so we don't double-schedule.
   429→    if (rafRef.current !== null) {
   430→      cancelAnimationFrame(rafRef.current);
   431→      rafRef.current = null;
   432→    }
   433→    if (spotlightOpen) {
   434→      rafRef.current = requestAnimationFrame(() => {
   435→        rafRef.current = null;
   436→        setIsDismissing(false);
   437→        setSelectedCardId(null);
   438→        setIsVisible(true);
   439→        cardTimerRef.current = setTimeout(() => setCardsVisible(true), 150);
   440→      });
   441→    } else {
   442→      rafRef.current = requestAnimationFrame(() => {
   443→        rafRef.current = null;
   444→        setIsVisible(false);
   445→        setCardsVisible(false);
   446→        setSelectedCardId(null);
   447→        setIsDismissing(false);
   448→        setMeasuredHeights(new Map());
   449→      });
   450→    }
   451→
   452→    return () => {
   453→      if (rafRef.current !== null) {
   454→        cancelAnimationFrame(rafRef.current);
   455→        rafRef.current = null;
   456→      }
   457→      if (cardTimerRef.current) {
   458→        clearTimeout(cardTimerRef.current);
   459→        cardTimerRef.current = null;
   460→      }
   461→    };
   462→  }, [spotlightOpen, element]);
   463→
   464→  // Cleanup all timers on unmount
   465→  useEffect(() => {
   466→    return () => {
   467→      if (dismissTimerRef.current) clearTimeout(dismissTimerRef.current);
   468→      if (selectTimerRef.current) clearTimeout(selectTimerRef.current);
   469→      if (cardTimerRef.current) clearTimeout(cardTimerRef.current);
   470→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current);
   471→    };
   472→  }, []);
   473→
   474→  // ── Dismiss with exit animation ────────────────────────────────────────
   475→
   476→  const dismissOverlay = useCallback(() => {
   477→    if (isDismissing) return;
   478→    setIsDismissing(true);
   479→    setCardsVisible(false);
   480→
   481→    dismissTimerRef.current = setTimeout(() => {
   482→      closeSpotlight();
   483→    }, 180);
   484→  }, [isDismissing, closeSpotlight]);
   485→
   486→  // ── Escape key ─────────────────────────────────────────────────────────
   487→
   488→  useEffect(() => {
   489→    if (!spotlightOpen || isDismissing) return;
   490→    const handleKeyDown = (e: KeyboardEvent) => {
   491→      if (e.key === 'Escape') {
   492→        e.preventDefault();
   493→        e.stopPropagation();
   494→        dismissOverlay();
   495→      }
   496→    };
   497→    window.addEventListener('keydown', handleKeyDown);
   498→    return () => window.removeEventListener('keydown', handleKeyDown);
   499→  }, [spotlightOpen, isDismissing, dismissOverlay]);
   500→
   501→  // ── Card click handler ─────────────────────────────────────────────────
   502→
   503→  const handleCardClick = useCallback((section: PropertySection) => {
   504→    if (isDismissing || selectedCardId) return;
   505→
   506→    const cardId = `${element.id}-${section.id}`;
   507→    const alreadyFloating = floatingCards.some(c => c.id === cardId);
   508→
   509→    // Don't allow clicking already-floating cards
   510→    if (alreadyFloating) return;
   511→
   512→    // Trigger selection animation
   513→    setSelectedCardId(section.id);
   514→
   515→    // After selection animation, create floating card and dismiss
   516→    selectTimerRef.current = setTimeout(() => {
   517→      const sectionIds = sections.map(s => s.id);
   518→      const { positions: cardPositions } = calculateCardPositions(
   519→        sectionIds,
   520→        dimensions.width || 1200,
   521→        dimensions.height || 800,
   522→        measuredHeights,
   523→      );
   524→      const idx = sections.findIndex(s => s.id === section.id);
   525→      const pos = cardPositions[idx] || { x: 100, y: 100 };
   526→
   527→      // Calculate target position for the floating card
   528→      const CARD_WIDTH = 320; // Use widest card width for consistent positioning
   529→      const PANEL_WIDTH = 288; // w-72 = 288px (properties panel)
   530→      const RIGHT_MARGIN = 90; // comfortable distance from right viewport edge
   531→      const PANEL_GAP = 24; // gap between card and properties panel
   532→      const TOP_OFFSET = 160; // comfortable distance from top (header + toolbar + breathing room)
   533→      const CARD_STACK_GAP = 40; // vertical gap between stacked cards
   534→
   535→      // In "both" or "panel" mode: card goes to the left of the properties panel
   536→      // In "floating" mode: card goes to the right side with comfortable margin from edge
   537→      const targetX = propertyDisplayMode !== 'floating'
   538→        ? window.innerWidth - PANEL_WIDTH - CARD_WIDTH - PANEL_GAP
   539→        : window.innerWidth - CARD_WIDTH - RIGHT_MARGIN;
   540→
   541→      // Stack cards vertically if there are already floating cards
   542→      const existingCards = floatingCards.length;
   543→      const targetY = TOP_OFFSET + existingCards * CARD_STACK_GAP;
   544→
   545→      const newCard: FloatingCardData = {
   546→        id: cardId,
   547→        sectionLabel: section.label,
   548→        x: targetX,
   549→        y: targetY,
   550→        minimized: false,
   551→        zIndex: Date.now(),
   552→        initialX: pos.x, // Start from spotlight card position
   553→        initialY: pos.y, // Start from spotlight card position
   554→      };
   555→      addFloatingCard(newCard);
   556→      dismissOverlay();
   557→    }, 280);
   558→  }, [isDismissing, selectedCardId, element.id, floatingCards, sections, dimensions, measuredHeights, addFloatingCard, dismissOverlay, propertyDisplayMode]);
   559→
   560→  // ── Early return if not open ───────────────────────────────────────────
   561→
   562→  if (!spotlightOpen) return null;
   563→
   564→  const { positions } = cardLayout;
   565→
   566→  // Effective opacity for the dim overlay — full dim, no cutout
   567→  const dimOpacity = isVisible && !isDismissing ? 1 : 0;
   568→  const dimTransition = isDismissing ? 'opacity 150ms ease-in' : 'opacity 200ms ease-out';
   569→
   570→  return (
   571→    <div
   572→      className="fixed inset-0 z-[99999]"
   573→      style={{ pointerEvents: isVisible ? 'auto' : 'none' }}
   574→    >
   575→      {/* ── Dim overlay — covers everything including the selected element ── */}
   576→      <div
   577→        className="absolute inset-0 bg-black/60 backdrop-blur-[2px]"
   578→        style={{
   579→          opacity: dimOpacity,
   580→          transition: dimTransition,
   581→        }}
   582→        onClick={dismissOverlay}
   583→      />
   584→
   585→      {/* ── Masonry property cards ───────────────────────────────────── */}
   586→      {sections.map((section, index) => {
   587→        const pos = positions[index];
   588→        if (!pos) return null;
   589→
   590→        const isHovered = hoveredCard === section.id;
   591→        const isSelected = selectedCardId === section.id;
   592→        const isUnselected = selectedCardId !== null && !isSelected;
   593→        const alreadyFloating = floatingCards.some(
   594→          c => c.id === `${element.id}-${section.id}`,
   595→        );
   596→
   597→        // Compute card opacity
   598→        let cardOpacity: number;
   599→        if (isUnselected) {
   600→          cardOpacity = 0;
   601→        } else if (isSelected) {
   602→          cardOpacity = 1;
   603→        } else if (cardsVisible && !isDismissing) {
   604→          cardOpacity = 1;
   605→        } else {
   606→          cardOpacity = 0;
   607→        }
   608→
   609→        // Compute card transform — subtle hover effect, NO size change for already-floating
   610→        let cardTransform: string;
   611→        if (isSelected) {
   612→          cardTransform = 'scale(1.02)';
   613→        } else if (alreadyFloating) {
   614→          cardTransform = 'scale(1)'; // Same size as normal cards
   615→        } else if (cardsVisible && !isDismissing && !isUnselected) {
   616→          cardTransform = isHovered ? 'translateY(-2px)' : 'translateY(0)';
   617→        } else {
   618→          cardTransform = 'scale(0.95)';
   619→        }
   620→
   621→        // Stagger delay (only for entrance, not for selection/dismiss)
   622→        const staggerDelay = (selectedCardId || isDismissing) ? 0 : index * 40;
   623→
   624→        // Transition timing
   625→        const cardTransition = isSelected
   626→          ? `all 200ms ease-in ${staggerDelay}ms`
   627→          : isUnselected
   628→          ? `all 180ms ease-in ${staggerDelay}ms`
   629→          : `all 250ms cubic-bezier(0.34, 1.56, 0.64, 1) ${staggerDelay}ms`;
   630→
   631→        // Section label for PropertyContent
   632→        const sectionLabel = sectionIdToLabel(section.id);
   633→
   634→        return (
   635→          <div
   636→            key={section.id}
   637→            ref={(el) => {
   638→              if (el) {
   639→                const height = el.offsetHeight;
   640→                if (height > 0 && measuredHeights.get(section.id) !== height) {
   641→                  setMeasuredHeights(prev => {
   642→                    const next = new Map(prev);
   643→                    next.set(section.id, height);
   644→                    return next;
   645→                  });
   646→                }
   647→              }
   648→            }}
   649→            className="absolute"
   650→            style={{
   651→              left: pos.x,
   652→              top: pos.y,
   653→              opacity: cardOpacity,
   654→              transform: cardTransform,
   655→              transition: cardTransition,
   656→              pointerEvents: isDismissing || isUnselected || alreadyFloating ? 'none' : 'auto',
   657→            }}
   658→            onMouseEnter={() => !isDismissing && !alreadyFloating && setHoveredCard(section.id)}
   659→            onMouseLeave={() => setHoveredCard(null)}
   660→            onClick={(e) => {
   661→              e.stopPropagation();
   662→              handleCardClick(section);
   663→            }}
   664→          >
   665→            <div
   666→              className={`${getSpotlightCardWidthClass(sectionLabel)} rounded-lg border overflow-hidden transition-all duration-200 ${
   667→                alreadyFloating
   668→                  ? 'bg-card/60 border-border/30 cursor-default'
   669→                  : isSelected
   670→                  ? 'bg-card border-primary/50 shadow-xl ring-2 ring-primary/30 cursor-pointer'
   671→                  : isHovered
   672→                  ? 'bg-card border-primary/30 shadow-lg cursor-pointer'
   673→                  : 'bg-card/95 border-border/50 shadow-md cursor-pointer'
   674→              }`}
   675→            >
   676→              {/* Title bar */}
   677→              <div
   678→                className={`flex items-center gap-1.5 px-2.5 py-1.5 border-b border-border/40 ${
   679→                  alreadyFloating
   680→                    ? 'bg-muted/20'
   681→                    : 'bg-muted/30'
   682→                }`}
   683→              >
   684→                <div className={`p-1 rounded shrink-0 ${
   685→                  alreadyFloating
   686→                    ? 'bg-muted/50 text-muted-foreground/50'
   687→                    : isSelected
   688→                    ? 'bg-primary/15 text-primary'
   689→                    : isHovered
   690→                    ? 'bg-primary/10 text-primary'
   691→                    : 'bg-muted text-muted-foreground'
   692→                }`}>
   693→                  {section.icon}
   694→                </div>
   695→                <span className={`text-[11px] font-medium truncate flex-1 ${
   696→                  alreadyFloating ? 'text-muted-foreground/50' : 'text-foreground'
   697→                }`}>
   698→                  {section.label}
   699→                </span>
   700→                {alreadyFloating && (
   701→                  <span className="text-[9px] text-primary/60 font-medium bg-primary/10 px-1.5 py-0.5 rounded">open</span>
   702→                )}
   703→              </div>
   704→
   705→              {/* Content — always show full content with clean padding */}
   706→              <div className="relative pointer-events-none select-none p-3">
   707→                <div style={{ opacity: alreadyFloating ? 0.3 : 0.5, filter: `grayscale(${alreadyFloating ? 80 : 50}%)` }}>
   708→                  <PropertyContent sectionLabel={sectionLabel} element={element} compact multiSelectIds={isMultiSelect ? multiSelectIds : undefined} />
   709→                </div>
   710→                <div
   711→                  className="absolute inset-0 pointer-events-none rounded-b-lg"
   712→                  style={{ backgroundColor: alreadyFloating ? 'rgba(59, 130, 246, 0.12)' : 'rgba(255, 255, 255, 0.35)' }}
   713→                />
   714→              </div>
   715→            </div>
   716→          </div>
   717→        );
   718→      })}
   719→
   720→      {/* ── Bottom hint ────────────────────────────────────────────────── */}
   721→      <div
   722→        className="absolute bottom-6 left-1/2 text-xs text-white/60"
   723→        style={{
   724→          opacity: cardsVisible && !isDismissing && !selectedCardId ? 1 : 0,
   725→          transform: `translateX(-50%) translateY(${cardsVisible && !isDismissing && !selectedCardId ? 0 : 8}px)`,
   726→          transition: 'opacity 250ms ease-out, transform 250ms ease-out',
   727→        }}
   728→      >
   729→        Press{' '}
   730→        <kbd className="px-1.5 py-0.5 rounded bg-white/10 text-white/80 font-mono text-[10px]">
   731→          Esc
   732→        </kbd>{' '}
   733→        to close · Click a card to pop it out
   734→      </div>
   735→    </div>
   736→  );
   737→}
