    63→// ── Re-ID element helper ────────────────────────────────────────────────────
    64→// When duplicating/pasting an element (especially a group), we must assign new
    65→// UUIDs to the element AND all of its children to avoid duplicate IDs in the
    66→// element tree.  Duplicate IDs cause lookups (findElementById, findParentGroup)
    67→// to return the wrong element.
    68→function reIdElement(element: CanvasElement): CanvasElement {
    69→  const result: CanvasElement = {
    70→    ...element,
    71→    id: uuidv4(),
    72→    // Deep-copy cornerPoints so duplicates/pastes have independent corners
    73→    // (a shallow copy would share the array reference with the source).
    74→    ...(element.cornerPoints
    75→      ? { cornerPoints: element.cornerPoints.map((p) => ({ x: p.x, y: p.y })) }
    76→      : {}),
    77→  };
    78→  if (result.type === 'group' && result.children) {
    79→    result.children = result.children.map((c) => reIdElement(c));
    80→  }
    81→  return result;
    82→}

    84→// ── Line–Rectangle intersection ────────────────────────────────────────────
    85→// Used by marquee selection to accurately test whether a line element
    86→// intersects the selection rectangle, rather than relying on the line's
    87→// bounding box which can be much larger than the visible line itself.
    88→
    89→/** Check if point (px, py) lies inside the axis-aligned rect [l,t,r,b]. */
    90→function pointInRect(px: number, py: number, l: number, t: number, r: number, b: number): boolean {
    91→  return px >= l && px <= r && py >= t && py <= b;
    92→}
    93→
    94→/** Return the cross product of vectors (p1→p2) × (p1→p3). */
    95→function cross(p1x: number, p1y: number, p2x: number, p2y: number, p3x: number, p3y: number): number {
    96→  return (p2x - p1x) * (p3y - p1y) - (p2y - p1y) * (p3x - p1x);
    97→}
    98→
    99→/**
   100→ * Check whether the line segment (x1,y1)→(x2,y2) intersects the
   101→ * axis-aligned rectangle [left, top, right, bottom].
   102→ *
   103→ * Algorithm:
   104→ *  1. If either endpoint is inside the rect → intersect.
   105→ *  2. Test intersection of the segment against each of the 4 rectangle
   106→ *     edges using the separating-axis (cross-product) test.
   107→ */
   108→function lineIntersectsRect(
   109→  x1: number, y1: number, x2: number, y2: number,
   110→  left: number, top: number, right: number, bottom: number,
   111→): boolean {
   112→  // Quick check: either endpoint inside the rect
   113→  if (pointInRect(x1, y1, left, top, right, bottom)) return true;
   114→  if (pointInRect(x2, y2, left, top, right, bottom)) return true;
   115→
   116→  // Check intersection with each of the 4 edges of the rectangle
   117→  // Top edge: (left,top)→(right,top)
   118→  if (segmentsIntersect(x1, y1, x2, y2, left, top, right, top)) return true;
   119→  // Bottom edge: (left,bottom)→(right,bottom)
   120→  if (segmentsIntersect(x1, y1, x2, y2, left, bottom, right, bottom)) return true;
   121→  // Left edge: (left,top)→(left,bottom)
   122→  if (segmentsIntersect(x1, y1, x2, y2, left, top, left, bottom)) return true;
   123→  // Right edge: (right,top)→(right,bottom)
   124→  if (segmentsIntersect(x1, y1, x2, y2, right, top, right, bottom)) return true;
   125→
   126→  return false;
   127→}
   128→
   129→/**
   130→ * Check whether two line segments (ax1,ay1)→(ax2,ay2) and
   131→ * (bx1,by1)→(bx2,by2) properly intersect (crossing each other,
   132→ * not just touching at endpoints).
   133→ */
   134→function segmentsIntersect(
   135→  ax1: number, ay1: number, ax2: number, ay2: number,
   136→  bx1: number, by1: number, bx2: number, by2: number,
   137→): boolean {
   138→  const d1 = cross(bx1, by1, bx2, by2, ax1, ay1);
   139→  const d2 = cross(bx1, by1, bx2, by2, ax2, ay2);
   140→  const d3 = cross(ax1, ay1, ax2, ay2, bx1, by1);
   141→  const d4 = cross(ax1, ay1, ax2, ay2, bx2, by2);
   142→
   143→  if (((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) &&
   144→      ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0))) {
   145→    return true;
   146→  }
   147→
   148→  // Collinear / endpoint-touching cases — treat as non-intersecting for
   149→  // marquee selection purposes (the endpoint-inside-rect check above
   150→  // already handles the case where a line endpoint is inside the rect).
   151→  return false;
   152→}

   154→// ── Undo / Redo ────────────────────────────────────────────────────────────
   155→const MAX_UNDO_HISTORY = 50;
   156→
   157→/**
   158→ * Depth counter for bulk operations that must produce a SINGLE undo entry.
   159→ * While > 0, pushUndo() becomes a no-op so batch actions (e.g. multi-element
   160→ * cut) can push one snapshot themselves instead of one per sub-operation.
   161→ */
   162→let suppressAutoUndoDepth = 0;

   164→/** Subset of store state that represents a "document" snapshot for undo/redo. */
   165→interface HistorySnapshot {
   166→  elements: CanvasElement[];
   167→  elementCounter: Record<ElementType, number>;
   168→  canvasSettings: CanvasSettings;
   169→}
   170→
   171→function takeSnapshot(state: {
   172→  elements: CanvasElement[];
   173→  elementCounter: Record<ElementType, number>;
   174→  canvasSettings: CanvasSettings;
   175→}): HistorySnapshot {
   176→  return {
   177→    elements: structuredClone(state.elements),
   178→    elementCounter: { ...state.elementCounter },
   179→    canvasSettings: structuredClone(state.canvasSettings),
   180→  };
   181→}
   182→
   183→// Debounce state for property update undo pushes.
   184→// Prevents flooding the undo stack when sliders or text inputs fire many rapid changes.
   185→let propertyUndoTimer: ReturnType<typeof setTimeout> | null = null;
   186→let pendingPropertySnapshot: HistorySnapshot | null = null;
   187→const PROPERTY_UNDO_DEBOUNCE_MS = 400;
   188→
   189→/** Internal helper: push a snapshot onto the undo stack and clear redo. */
   190→function commitSnapshotToUndo(snapshot: HistorySnapshot) {
   191→  const current = useDesignerStore.getState();
   192→  const newUndoStack = [...current.undoStack, snapshot];
   193→  if (newUndoStack.length > MAX_UNDO_HISTORY) newUndoStack.shift();
   194→  useDesignerStore.setState({
   195→    undoStack: newUndoStack,
   196→    redoStack: [],
   197→    canUndo: true,
   198→    canRedo: false,
   199→  });
   200→}
   201→
   202→/** Push a debounced undo snapshot for property changes.
   203→ *  The first call captures the "before" state, immediately clears the redo
   204→ *  stack and updates canUndo/canRedo so the UI stays consistent. Subsequent
   205→ *  calls within the debounce window are absorbed. When the timer fires, the
   206→ *  snapshot is pushed onto the undo stack. */
   207→function debouncedPropertyPushUndo(
   208→  state: { elements: CanvasElement[]; elementCounter: Record<ElementType, number>; canvasSettings: CanvasSettings },
   209→) {
   210→  if (propertyUndoTimer === null) {
   211→    // First call in a new interaction — capture the BEFORE state immediately
   212→    pendingPropertySnapshot = takeSnapshot(state);
   213→    // Immediately clear redo stack & update flags so the UI is consistent
   214→    // from the moment the first property change happens.
   215→    useDesignerStore.setState({
   216→      redoStack: [],
   217→      canRedo: false,
   218→      canUndo: true, // the pending snapshot guarantees at least one undo step
   219→    });
   220→  }
   221→  // Reset the debounce timer
   222→  if (propertyUndoTimer) clearTimeout(propertyUndoTimer);
   223→  propertyUndoTimer = setTimeout(() => {
   224→    if (pendingPropertySnapshot !== null) {
   225→      // Push the captured "before" snapshot
   226→      commitSnapshotToUndo(pendingPropertySnapshot);
   227→      pendingPropertySnapshot = null;
   228→    }
   229→    propertyUndoTimer = null;
   230→  }, PROPERTY_UNDO_DEBOUNCE_MS);
   231→}
   232→
   233→/** Immediately flush any pending debounced undo entry (call before non-debounced actions).
   234→ *  Returns true if a pending snapshot was committed. */
   235→function flushPropertyUndo(): boolean {
   236→  if (propertyUndoTimer !== null) {
   237→    clearTimeout(propertyUndoTimer);
   238→    if (pendingPropertySnapshot !== null) {
   239→      commitSnapshotToUndo(pendingPropertySnapshot);
   240→      pendingPropertySnapshot = null;
   241→    }
   242→    propertyUndoTimer = null;
   243→    return true;
   244→  }
   245→  return false;
   246→}
   247→
   248→/** Cancel any pending debounced undo entry WITHOUT committing it.
   249→ *  Used when undo/redo is invoked — the pending "before" state is already
   250→ *  what the user wants to go back to, so we don't need to push it. */
   251→function cancelPropertyUndo() {
   252→  if (propertyUndoTimer !== null) {
   253→    clearTimeout(propertyUndoTimer);
   254→    pendingPropertySnapshot = null;
   255→    propertyUndoTimer = null;
   256→    // The optimistic `canUndo: true` set when the pending snapshot was created
   257→    // must be recomputed — if the stack was empty (first-ever edit) the Undo
   258→    // button would otherwise stay enabled while undo() is a no-op.
   259→    useDesignerStore.setState({ canUndo: useDesignerStore.getState().undoStack.length > 0 });
   260→  }
   261→}

   263→/**
   264→ * Pure recursive helper: returns a NEW elements tree with the properties of the
   265→ * element identified by `id` replaced. Searches both top-level elements and the
   266→ * `children` arrays of groups (arbitrarily deep, matching `findElementById`).
   267→ *
   268→ * Unlike the `updateElementProperties` store action, this is side-effect-free —
   269→ * no undo snapshot is captured — so it is safe to call on every mousemove of a
   270→ * drag whose undo entry was already pushed at drag start (e.g. table resize).
   271→ */
   272→function setElementPropertiesInTree(
   273→  elements: CanvasElement[],
   274→  id: string,
   275→  properties: ElementProperties,
   276→): CanvasElement[] {
   277→  const updateChildren = (children: CanvasElement[]): CanvasElement[] => {
   278→    const idx = children.findIndex((c) => c.id === id);
   279→    if (idx !== -1) {
   280→      const next = [...children];
   281→      next[idx] = { ...next[idx], properties };
   282→      return next;
   283→    }
   284→    // Recurse into nested groups (groups are normally single-level here, but
   285→    // recurse anyway to stay consistent with findElementById).
   286→    let changed = false;
   287→    const maybeDeeper = children.map((c) => {
   288→      if (c.type === 'group' && c.children) {
   289→        const newGrand = updateChildren(c.children);
   290→        if (newGrand !== c.children) {
   291→          changed = true;
   292→          return { ...c, children: newGrand };
   293→        }
   294→      }
   295→      return c;
   296→    });
   297→    return changed ? maybeDeeper : children;
   298→  };
   299→
   300→  let changed = false;
   301→  const result = elements.map((el) => {
   302→    if (el.id === id) {
   303→      changed = true;
   304→      return { ...el, properties };
   305→    }
   306→    if (el.type === 'group' && el.children) {
   307→      const newChildren = updateChildren(el.children);
   308→      if (newChildren !== el.children) {
   309→        changed = true;
   310→        return { ...el, children: newChildren };
   311→      }
   312→    }
   313→    return el;
   314→  });
   315→  return changed ? result : elements;
   316→}

   827→/**
   828→ * Translate an element by (dx, dy): moves x/y and keeps line endpoints and
   829→ * absolute rectangle corner points in sync. Pure — returns a new element.
   830→ * Replaces the per-action boilerplate that was previously copy-pasted at
   831→ * every call site that shifted an element wholesale.
   832→ */
   833→function translateElement(el: CanvasElement, dx: number, dy: number): CanvasElement {
   834→  if (dx === 0 && dy === 0) return el;
   835→  const moved: CanvasElement = { ...el, x: el.x + dx, y: el.y + dy };
   836→  if (el.type === 'line' && el.lineStartX !== undefined) {
   837→    moved.lineStartX = el.lineStartX + dx;
   838→    moved.lineStartY = (el.lineStartY ?? 0) + dy;
   839→    moved.lineEndX = (el.lineEndX ?? 0) + dx;
   840→    moved.lineEndY = (el.lineEndY ?? 0) + dy;
   841→  }
   842→  if (el.cornerPoints) {
   843→    moved.cornerPoints = el.cornerPoints.map((p) => ({ x: p.x + dx, y: p.y + dy }));
   844→  }
   845→  return moved;
   846→}

   848→/** Move an element, correctly adjusting line endpoint / corner-point deltas when x or y changes. */
   849→function moveElementWithLineDelta(el: CanvasElement, updates: { x?: number; y?: number }): CanvasElement {
   850→  // Compute BOTH axis deltas up front so a combined {x, y} update never drops
   851→  // the y (or x) adjustment — previously each branch returned early.
   852→  if (el.type === 'line' && (el.lineStartX !== undefined || el.lineStartY !== undefined) &&
   853→      (updates.x !== undefined || updates.y !== undefined)) {
   854→    const dx = updates.x !== undefined ? updates.x - el.x : 0;
   855→    const dy = updates.y !== undefined ? updates.y - el.y : 0;
   856→    return {
   857→      ...el,
   858→      ...updates,
   859→      lineStartX: el.lineStartX !== undefined ? el.lineStartX + dx : el.lineStartX,
   860→      lineStartY: el.lineStartY !== undefined ? el.lineStartY + dy : el.lineStartY,
   861→      lineEndX: el.lineEndX !== undefined ? el.lineEndX + dx : el.lineEndX,
   862→      lineEndY: el.lineEndY !== undefined ? el.lineEndY + dy : el.lineEndY,
   863→    };
   864→  }
   865→  // Translate rectangle corner points (absolute) when the element moves.
   866→  if (el.cornerPoints && (updates.x !== undefined || updates.y !== undefined)) {
   867→    const dx = (updates.x ?? el.x) - el.x;
   868→    const dy = (updates.y ?? el.y) - el.y;
   869→    if (dx !== 0 || dy !== 0) {
   870→      return {
   871→        ...el,
   872→        ...updates,
   873→        cornerPoints: el.cornerPoints.map((p) => ({ x: p.x + dx, y: p.y + dy })),
   874→      };
   875→    }
   876→  }
   877→  return { ...el, ...updates };
   878→}

   886→export const useDesignerStore = create<DesignerState>((set, get) => {
   887→  /**
   888→   * Shared implementation for pasteElement (offset +20/+20) and
   889→   * pasteElementInPlace (offset 0/0). Previously these were two ~95-line
   890→   * near-identical action bodies; only the paste offset differed.
   891→   */
   892→  const pasteClones = (offsetX: number, offsetY: number) => {
   893→    invalidateSnapCache();
   894→    const state = get();
   895→
   896→    // Prefer multi-element clipboard; fall back to legacy single-element one
   897→    const clips = state.clipboardElements.length > 0
   898→      ? state.clipboardElements
   899→      : state.clipboardElement
   900→        ? [state.clipboardElement]
   901→        : [];
   902→    if (clips.length === 0) return;
   903→
   904→    state.pushUndo();
   905→    const counter = { ...state.elementCounter };
   906→    const newElements: CanvasElement[] = [];
   907→    const newSelectionIds: string[] = [];
   908→
   909→    for (const clipEl of clips) {
   910→      counter[clipEl.type] += 1;
   911→      const reIded = reIdElement(clipEl);
   912→      const newEl: CanvasElement = {
   913→        ...reIded,
   914→        x: clipEl.x + offsetX,
   915→        y: clipEl.y + offsetY,
   916→        name: getDefaultName(clipEl.type, counter[clipEl.type]),
   917→        // Offset absolute rectangle corner points by the same paste offset
   918→        ...(reIded.cornerPoints
   919→          ? { cornerPoints: reIded.cornerPoints.map((p) => ({ x: p.x + offsetX, y: p.y + offsetY })) }
   920→          : {}),
   921→      };
   922→      newElements.push(newEl);
   923→      newSelectionIds.push(newEl.id);
   924→    }
   925→
   926→    if (state.editingGroupId) {
   927→      set({
   928→        elements: state.elements.map((el) => {
   929→          if (el.id === state.editingGroupId && el.type === 'group' && el.children) {
   930→            return { ...el, children: [...el.children, ...newElements] };
   931→          }
   932→          return el;
   933→        }),
   934→        selectedElementIds: newSelectionIds,
   935→        selectedElementId: deriveSelectedElementId(newSelectionIds),
   936→        elementCounter: counter,
   937→        isDirty: true,
   938→      });
   939→      queueMicrotask(() => get().recalculateGroupBounds(state.editingGroupId!));
   940→    } else {
   941→      set({
   942→        elements: [...state.elements, ...newElements],
   943→        selectedElementIds: newSelectionIds,
   944→        selectedElementId: deriveSelectedElementId(newSelectionIds),
   945→        elementCounter: counter,
   946→        isDirty: true,
   947→      });
   948→    }
   949→  };

   963→          }
   964→        }
