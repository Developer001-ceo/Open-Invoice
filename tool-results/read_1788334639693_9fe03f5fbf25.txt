    42→// Resize handle component
    43→function ResizeHandle({ position, rotation, onResizeStart }: { position: string; rotation: number; onResizeStart: (handle: string, e: React.MouseEvent) => void }) {
    44→  const isCorner = position.length === 2;
    45→  const size = isCorner ? 'w-2.5 h-2.5' : 'w-2 h-2';
    46→
    47→  // Rotation-aware cursor: at 0° (and 45° increments) this returns the native
    48→  // built-in CSS cursor; at other angles it generates a custom SVG double-arrow
    49→  // rotated to the handle's visual screen direction. See src/lib/resize-cursor.ts.
    50→  const cursor = getResizeCursor(position, rotation);
    51→
    52→  const getPositionStyle = () => {
    53→    const offset = -5;
    54→    switch (position) {
    55→      case 'nw': return { top: offset, left: offset };
    56→      case 'n': return { top: offset, left: '50%', transform: 'translateX(-50%)' };
    57→      case 'ne': return { top: offset, right: offset };
    58→      case 'e': return { top: '50%', right: offset, transform: 'translateY(-50%)' };
    59→      case 'se': return { bottom: offset, right: offset };
    60→      case 's': return { bottom: offset, left: '50%', transform: 'translateX(-50%)' };
    61→      case 'sw': return { bottom: offset, left: offset };
    62→      case 'w': return { top: '50%', left: offset, transform: 'translateY(-50%)' };
    63→      default: return {};
    64→    }
    65→  };
    66→
    67→  // Accessible label for screen readers + arrow-key nudging for keyboard users.
    68→  const ariaLabel = `${position} resize handle`;
    69→  const handleKeyDown = (e: React.KeyboardEvent) => {
    70→    if (e.key === 'Enter' || e.key === ' ') {
    71→      e.preventDefault();
    72→      // Keyboard users can't drag, but Enter/Space focuses the element for
    73→      // subsequent arrow-key nudging handled by the canvas keyboard layer.
    74→    }
    75→  };
    76→
    77→  return (
    78→    <div
    79→      data-resize-handle
    80→      role="button"
    81→      tabIndex={0}
    82→      aria-label={ariaLabel}
    83→      className={`absolute ${size} bg-white border-2 border-primary rounded-sm hover:bg-primary hover:border-primary transition-colors focus:outline-none focus:ring-2 focus:ring-primary/60`}
    84→      style={{ ...getPositionStyle() as React.CSSProperties, cursor, zIndex: 10000, pointerEvents: 'auto' }}
    85→      onMouseDown={(e) => {
    86→        if (e.button === 1) {
    87→          e.preventDefault();
    88→          useDesignerStore.getState().startPanning(e.clientX, e.clientY);
    89→          return;
    90→        }
    91→        if (e.button !== 0) return;
    92→        e.stopPropagation();
    93→        onResizeStart(position, e);
    94→      }}
    95→      onKeyDown={handleKeyDown}
    96→    />
    97→  );
    98→}

   151→// Rectangle corner handle — appears at each of the 4 corner points of a
   152→// selected rectangle. Behavior is ALT-gated:
   153→//  - Alt+drag  → individual corner deformation (the corner moves independently,
   154→//    the other 3 stay fixed). On a fresh axis-aligned rect this lazily
   155→//    converts it to a polygon; on an already-deformed rect it just moves the
   156→//    dragged corner.
   157→//  - Plain drag (no Alt) → standard corner resize (nw/ne/se/sw). For a normal
   158→//    rect this preserves the axis-aligned box; for a deformed rect the polygon
   159→//    resizes per-edge (the opposite corner stays fixed).
   160→// Alignment modifiers (held during an Alt+drag). The DRAGGED corner is the one
   161→// that moves into alignment; the partner corner(s) stay fixed:
   162→//  - Alt+Shift             → horizontal align (dragged corner's Y snaps to its
   163→//                            horizontal partner's Y).
   164→//  - Alt+Ctrl              → vertical align (dragged corner's X snaps to its
   165→//                            vertical partner's X).
   166→//  - Alt+Shift+Ctrl        → both (dragged corner snaps to (vPartner.X, hPartner.Y)).
   167→function CornerHandle({
   168→  x,
   169→  y,
   170→  cornerIndex,
   171→  isCustom,
   172→  rotation,
   173→  onCornerMouseDown,
   174→}: {
   175→  x: number;
   176→  y: number;
   177→  cornerIndex: number; // 0=TL, 1=TR, 2=BR, 3=BL
   178→  isCustom: boolean; // whether the rectangle already has deformed corners
   179→  rotation: number; // rectangle rotation in degrees (for rotation-aware cursor)
   180→  onCornerMouseDown: (cornerIndex: number, altHeld: boolean, e: React.MouseEvent) => void;
   181→}) {
   182→  // Rotation-aware cursor: maps corner index to its handle position (nw/ne/se/sw)
   183→  // and generates a cursor rotated to the visual screen direction. At 0° this
   184→  // returns the native nwse/nesw built-in; at other angles a custom SVG arrow.
   185→  const handlePos = CORNER_HANDLE_MAP[cornerIndex];
   186→  const cursor = getResizeCursor(handlePos, rotation);
   187→  return (
   188→    <div
   189→      data-resize-handle
   190→      data-corner-handle
   191→      className={`absolute w-2.5 h-2.5 rounded-sm transition-colors ${
   192→        isCustom
   193→          ? 'bg-white border-2 border-purple-500 hover:bg-purple-500 hover:border-purple-500'
   194→          : 'bg-white border-2 border-primary hover:bg-primary hover:border-primary'
   195→      }`}
   196→      style={{
   197→        left: x - 5,
   198→        top: y - 5,
   199→        cursor,
   200→        zIndex: 10000,
   201→        pointerEvents: 'auto',
   202→      }}
   203→      title={isCustom ? 'Alt+drag to move corner · Alt+Shift horizontal align · Alt+Ctrl vertical align · Alt+Shift+Ctrl both' : 'Alt+drag to move corner independently'}
   204→      onMouseDown={(e) => {
   205→        if (e.button === 1) {
   206→          e.preventDefault();
   207→          useDesignerStore.getState().startPanning(e.clientX, e.clientY);
   208→          return;
   209→        }
   210→        if (e.button !== 0) return;
   211→        e.stopPropagation();
   212→        onCornerMouseDown(cornerIndex, e.altKey, e);
   213→      }}
   214→    />
   215→  );
   216→}

   227→
   228→  const isEditing = editingTextElementId === element.id;
   229→
   230→  const handleDoubleClick = useCallback((e: React.MouseEvent) => {
   231→    e.stopPropagation();
   232→    // Select the element and open the inline text editor on canvas
   233→    selectElement(element.id);
   234→    setEditingTextElement(element.id);
   235→

   237→    // formatting tools while editing (keyboard shortcuts like Shift+S
   238→    // produce characters instead of triggering shortcuts in edit mode).
   239→    const store = useDesignerStore.getState();
   240→    const cardId = `${element.id}-text-content`;
   241→    const alreadyFloating = store.floatingCards.some(c => c.id === cardId);
   242→    if (!alreadyFloating) {
   243→      const CARD_WIDTH = 320;
   244→      const PANEL_WIDTH = 288;
   245→      const PANEL_GAP = 24;
   246→      const RIGHT_MARGIN = 90;
   247→      const TOP_OFFSET = 160;
   248→      const CARD_STACK_GAP = 40;
   249→

   251→      const targetX = propertyDisplayMode !== 'floating'
   252→        ? window.innerWidth - PANEL_WIDTH - CARD_WIDTH - PANEL_GAP
   253→        : window.innerWidth - CARD_WIDTH - RIGHT_MARGIN;
   254→
   255→      const existingCards = store.floatingCards.length;
   256→      const targetY = TOP_OFFSET + existingCards * CARD_STACK_GAP;

   259→        id: cardId,
   260→        sectionLabel: 'Formatting',
   261→        x: targetX,

   282→      _lastTextScrollTop = tiptapEl.scrollTop;

   284→    setEditingTextElement(null);
   285→  }, [setEditingTextElement]);
   286→
   287→  return (
   288→    <div
   289→      className={`w-full h-full outline-none overflow-hidden ${
   290→        isEditing
   291→          ? 'ring-2 ring-primary/50 rounded cursor-default'
   292→          : ''
   293→      }`}
   294→      onDoubleClick={handleDoubleClick}
   295→    >
   296→      {isEditing ? (
   297→        <CanvasTextEditor
   298→          elementId={element.id}
   299→          props={props}
   300→          onContentChange={handleContentChange}
   301→          onFinishEditing={handleFinishEditing}
   302→        />
   303→      ) : (
   304→        <RichTextRenderer props={props} initialScrollTop={_lastTextScrollTop} />
   305→      )}
   306→    </div>
   307→  );

   310→// Table Element Renderer — uses CSS Grid for fixed row heights
   311→const TableElementRenderer = memo(function TableElementRenderer({ element }: { element: CanvasElement }) {
   312→  const rawProps = (element.properties as { type: 'table'; data: TableProperties }).data;
   313→  const updateElementProperties = useDesignerStore((s) => s.updateElementProperties);
   314→
   315→  // Alt-key cell selection state from the store
   316→  const altHeld = useDesignerStore((s) => s.altHeld);
   317→  const selectedTableCells = useDesignerStore((s) => s.selectedTableCells);
   318→  const hoveredTableCell = useDesignerStore((s) => s.hoveredTableCell);
   319→  const isTableCellSelecting = useDesignerStore((s) => s.isTableCellSelecting);
   320→  // The table that owns the current cell selection. The selection is scoped
   321→  // to this table only — other tables on the canvas must NOT highlight cells
   322→  // just because they share the same {row, col} coordinates.
   323→  const tableCellSelectTableId = useDesignerStore((s) => s.tableCellSelectTableId);
   324→  const isTableSelected = useDesignerStore(
   325→    useCallback((s) => s.selectedElementIds.includes(element.id), [element.id])
   326→  );
   327→  // Whether this table is the active owner of the cell selection.
   328→  const selectionOwnsTable = tableCellSelectTableId === element.id;
   329→
   330→  // Normalize legacy table properties using shared helper
   331→  const props: TableProperties = normalizeTableProps(rawProps);
   332→  const [editingCell, setEditingCell] = useState<{ r: number; c: number } | null>(null);
   333→  const [editValue, setEditValue] = useState('');
   334→
   335→  const handleCellDoubleClick = (r: number, c: number) => {
   336→    // Entering edit mode — clear any cell selection first
   337→    useDesignerStore.getState().clearTableCellSelection();
   338→    setEditingCell({ r, c });
   339→    setEditValue(props.cellData[r]?.[c] || '');
   340→  };
   341→
   342→  const handleCellBlur = () => {
   343→    if (editingCell) {
   344→      const newCellData = props.cellData.map((row) => [...row]);
   345→      newCellData[editingCell.r][editingCell.c] = editValue;
   346→      updateElementProperties(element.id, {
   347→        type: 'table',
   348→        data: { ...props, cellData: newCellData },
   349→      });
   350→      setEditingCell(null);
   351→    }
   352→  };
   353→
   354→  // ── Alt-key cell selection handlers ──────────────────────────────────────
   355→  const handleCellMouseEnter = useCallback((r: number, c: number) => {
   356→    if (!altHeld) return;
   357→    const state = useDesignerStore.getState();
   358→    state.setHoveredTableCell({ row: r, col: c, tableId: element.id });
   359→
   360→    // If currently dragging (Alt+drag multi-select), update the selection range
   361→    if (state.isTableCellSelecting) {
   362→      state.updateTableCellSelecting(r, c);
   363→    }
   364→  }, [altHeld, element.id]);
   365→
   366→  const handleCellMouseLeave = useCallback(() => {
   367→    if (!altHeld) return;
   368→    useDesignerStore.getState().setHoveredTableCell(null);
   369→  }, [altHeld]);
   370→
   371→  const handleCellContextMenu = useCallback((e: React.MouseEvent, r: number, c: number) => {
   372→    if (!altHeld) return;
   373→    e.preventDefault();
   374→    e.stopPropagation();
   375→    const state = useDesignerStore.getState();
   376→    // Alt+right-click always toggles the cell selection
   377→    state.selectTableCell(element.id, r, c, true);
   378→  }, [altHeld, element.id]);
   379→
   380→  const handleCellPointerDown = useCallback((e: React.PointerEvent, r: number, c: number) => {
   381→    if (e.button !== 0) return;
   382→    const state = useDesignerStore.getState();
   383→
   384→    if (altHeld) {
   385→      e.stopPropagation();
   386→      e.preventDefault();
   387→      if (!state.isTableCellSelecting) {
   388→        state.startTableCellSelecting(element.id, r, c, state.selectedTableCells);
   389→      }
   390→    } else {
   391→      // Non-Alt click — only deselect if clicking an unselected cell
   392→      // Clicking a selected cell keeps the selection (double-click enters edit mode instead)
   393→      if (state.selectedTableCells.length > 0) {
   394→        const cellKey = `${r}-${c}`;
   395→        const isAlreadySelected = state.selectedTableCells.some(c => `${c.row}-${c.col}` === cellKey);
   396→        if (!isAlreadySelected) {
   397→          e.stopPropagation();
   398→          e.preventDefault();
   399→          state.clearTableCellSelection();
   400→        }
   401→      }
   402→    }
   403→  }, [altHeld, element.id]);
   404→
   405→  const handleCellPointerUp = useCallback(() => {
   406→    const state = useDesignerStore.getState();
   407→    if (state.isTableCellSelecting) {
   408→      state.finishTableCellSelecting();
   409→    }
   410→  }, []);
   411→
   412→  // Cell styling and border helpers are now imported from table-helpers.ts
   413→  // They are used as: getCellBg(r, c, props), getCellBorderTop(r, c, props), etc.
   414→
   415→  const bw = props.borderWidth;
   416→  const borderStyleValue = props.borderStyle;
   417→  const borderColorValue = props.borderColor;
   418→  // Collapsed border model: each shared edge is drawn only ONCE.
   419→  // Every cell draws its top and left borders.
   420→  // Only the last row draws bottom borders; only the last column draws right borders.
   421→  // This guarantees exactly one border per edge, all at the same thickness.
   422→  const borderStr = `${bw}px ${borderStyleValue} ${borderColorValue}`;
   423→
   424→  // Use colWidths/rowHeights fractions for grid template.
   425→  // We convert fractions to EXPLICIT PIXEL values rather than using CSS `fr`
   426→  // units. With `fr`, the browser redistributes all track sizes whenever any
   427→  // fraction changes, and sub-pixel rounding can cause other rows/columns to
   428→  // shift by ~1px during a resize drag. With explicit `px` values, each
   429→  // track's size is computed independently and stays stable when other tracks
   430→  // change — only the two resizing tracks actually change size. The last
   431→  // track absorbs any rounding remainder so the grid fills the element exactly.
   432→  const cw = ensureColWidths(props.cols, props.colWidths);
   433→  const rh = ensureRowHeights(props.rows, props.rowHeights);
   434→  const totalColFr = cw.reduce((a, b) => a + b, 0) || 1;
   435→  const visibleRowHeights = props.showHeader ? rh : rh.slice(1);
   436→  const totalRowFr = (visibleRowHeights.length > 0 ? visibleRowHeights : [1]).reduce((a, b) => a + b, 0) || 1;
   437→
   438→  // Compute explicit pixel sizes; the last track gets the remainder to avoid
   439→  // gaps/overflow from rounding.
   440→  const computeTrackSizes = (fracs: number[], total: number, totalFr: number): string[] => {
   441→    if (fracs.length === 0) return ['100%'];
   442→    const sizes = fracs.map(f => (f / totalFr) * total);
   443→    // Last track absorbs the rounding difference
   444→    const allocated = sizes.slice(0, -1).reduce((a, b) => a + b, 0);
   445→    sizes[sizes.length - 1] = total - allocated;
   446→    return sizes.map(s => `${s}px`);
   447→  };
   448→
   449→  const gridTemplateColumns = computeTrackSizes(cw, element.width, totalColFr).join(' ');
   450→  const gridTemplateRows = computeTrackSizes(
   451→    visibleRowHeights.length > 0 ? visibleRowHeights : [1],
   452→    element.height,
   453→    totalRowFr
   454→  ).join(' ');
   455→
   456→  // Build a Set of selected cell keys for quick lookup.
   457→  // IMPORTANT: only include cells when THIS table is the owner of the
   458→  // selection (tableCellSelectTableId === element.id). Without this guard,
   459→  // every table on the canvas would highlight cells at the same {row, col}
   460→  // as the selected ones, making it look like all tables got selected.
   461→  const selectedCellSet = useMemo(() => {
   462→    const set = new Set<string>();
   463→    if (!selectionOwnsTable) return set;
   464→    for (const cell of selectedTableCells) {
   465→      set.add(`${cell.row}-${cell.col}`);
   466→    }
   467→    return set;
   468→  }, [selectedTableCells, selectionOwnsTable]);
   469→
   470→  // Render all visible cells as grid children
   471→  const renderCells = () => {
   472→    const cells: React.ReactNode[] = [];
   473→
   474→    for (let r = 0; r < props.rows; r++) {
   475→      // Skip header row if hidden
   476→      if (r === 0 && !props.showHeader) continue;
   477→
   478→      const gridRow = props.showHeader ? r + 1 : r; // 1-based grid row
   479→
   480→      for (let c = 0; c < props.cols; c++) {
   481→        const padding = getCellPadding(r, c, props);
   482→        const bgColor = getCellBg(r, c, props);
   483→        const textColor = getCellColor(r, c, props);
   484→        const fontFamily = getCellFontFamily(r, c, props);
   485→        const fontWeight = getCellFontWeight(r, c, props);
   486→        const fontSize = getCellFontSize(r, c, props);
   487→        const textAlign = getCellTextAlign(r, c, props);
   488→        const verticalAlign = getCellVerticalAlign(r, c, props);
   489→        const textTransform = getCellTextTransform(r, c, props);
   490→        const rotation = getCellRotation(r, c, props);
   491→
   492→        // ── Alt-key cell selection visual feedback ──────────────────────────
   493→        const cellKey = `${r}-${c}`;
   494→        const isSelected = selectedCellSet.has(cellKey);
   495→        const isHovered = altHeld && hoveredTableCell?.row === r && hoveredTableCell?.col === c && hoveredTableCell?.tableId === element.id;
   496→
   497→        // Collapsed border model:
   498→        // - Every cell draws top & left borders (handles outer + inner edges)
   499→        // - Only last row draws bottom border (outer bottom edge)
   500→        // - Only last column draws right border (outer right edge)
   501→        // This avoids double-thick inner borders since each edge is drawn exactly once.
   502→        // For selected cells, borders stay normal — only the dim overlay marks selection.
   503→        const borderTop = getCellBorderTop(r, c, props) ? borderStr : 'none';
   504→        const borderLeft = getCellBorderLeft(r, c, props) ? borderStr : 'none';
   505→        const borderBottom = (r === props.rows - 1 && getCellBorderBottom(r, c, props)) ? borderStr : 'none';
   506→        const borderRight = (c === props.cols - 1 && getCellBorderRight(r, c, props)) ? borderStr : 'none';
   507→
   508→        // Corner radius: apply to corner cells so borders follow the curve
   509→        const isFirstVisibleRow = props.showHeader ? r === 0 : r === 1;
   510→        const isLastRow = r === props.rows - 1;
   511→        const isFirstCol = c === 0;
   512→        const isLastCol = c === props.cols - 1;
   513→        const cellBorderRadius = `${isFirstVisibleRow && isFirstCol ? radii.topLeft : 0}px ${isFirstVisibleRow && isLastCol ? radii.topRight : 0}px ${isLastRow && isLastCol ? radii.bottomRight : 0}px ${isLastRow && isFirstCol ? radii.bottomLeft : 0}px`;
   514→
   515→        // Vertical alignment via flexbox
   516→        const justifyContent = verticalAlign === 'top'
   517→          ? 'flex-start'
   518→          : verticalAlign === 'bottom'
   519→            ? 'flex-end'
   520→            : 'center';
   521→
   522→        // For 'capitalize', we apply proper title case via JS (first letter uppercase, rest lowercase)
   523→        // because CSS text-transform: capitalize only uppercases first letters without lowercasing the rest.
   524→        const effectiveTextTransform = textTransform === 'capitalize' ? 'none' : textTransform;
   525→        const cellText = props.cellData[r]?.[c] || '';
   526→        const displayText = textTransform === 'capitalize' ? properCapitalize(cellText) : cellText;

   536→              backgroundColor: bgColor,
   537→              color: textColor,
   538→              fontFamily,
   539→              fontWeight,
   540→              fontSize: `${fontSize}px`,

   542→              ...(textAlign === 'justify' ? { textAlignLast: 'left' as const, textJustify: 'inter-character' as const, hyphens: 'auto' as const } : {}),
   543→              textTransform: effectiveTextTransform as 'none' | 'uppercase' | 'lowercase',
   544→              borderTop,
   545→              borderBottom,
   546→              borderLeft,
   547→              borderRight,

   551→              display: 'flex',
   552→              alignItems: justifyContent,
   553→              minHeight: 0, // critical: allows grid to shrink below content size
   554→              minWidth: 0,
   555→              ...(rotation ? { transform: `rotate(${rotation}deg)` } : {}),

   557→              cursor: altHeld && isTableSelected ? 'pointer' : undefined,

   589→                  fontSize: 'inherit',

   591→                  color: 'inherit',

   604→                  lineHeight: 1.2,
   605→                  position: 'relative',
   606→                  zIndex: 2,
   607→                }}
   608→              >
   609→                {displayText}
   610→              </span>
   611→            )}
   612→          </div>
   613→        );
   614→      }
   615→    }
   616→
   617→    return cells;
   618→  };
   619→
   620→  const radii = resolveCornerRadius(props.cornerRadius);
   621→
   622→  return (
   623→    <div
   624→      className="w-full h-full"
   625→      style={{
   626→        opacity: props.opacity,
   627→        borderRadius: `${radii.topLeft}px ${radii.topRight}px ${radii.bottomRight}px ${radii.bottomLeft}px`,
   628→      }}
   629→    >
   630→      <div
   631→        style={{

   638→          overflow: 'hidden',

   640→          userSelect: altHeld && isTableSelected ? 'none' : undefined,
   641→        }}
   642→      >
   643→        {renderCells()}
   644→      </div>
   645→    </div>
   646→  );
