    42→    42→// Resize handle component
    43→    43→function ResizeHandle({ position, rotation, onResizeStart }: { position: string; rotation: number; onResizeStart: (handle: string, e: React.MouseEvent) => void }) {
    44→    44→  const isCorner = position.length === 2;
    45→    45→  const size = isCorner ? 'w-2.5 h-2.5' : 'w-2 h-2';
    46→    46→
    47→    47→  // Rotation-aware cursor: at 0° (and 45° increments) this returns the native
    48→    48→  // built-in CSS cursor; at other angles it generates a custom SVG double-arrow
    49→    49→  // rotated to the handle's visual screen direction. See src/lib/resize-cursor.ts.
    50→    50→  const cursor = getResizeCursor(position, rotation);
    51→    51→
    52→    52→  const getPositionStyle = () => {
    53→    53→    const offset = -5;
    54→    54→    switch (position) {
    55→    55→      case 'nw': return { top: offset, left: offset };
    56→    56→      case 'n': return { top: offset, left: '50%', transform: 'translateX(-50%)' };
    57→    57→      case 'ne': return { top: offset, right: offset };
    58→    58→      case 'e': return { top: '50%', right: offset, transform: 'translateY(-50%)' };
    59→    59→      case 'se': return { bottom: offset, right: offset };
    60→    60→      case 's': return { bottom: offset, left: '50%', transform: 'translateX(-50%)' };
    61→    61→      case 'sw': return { bottom: offset, left: offset };
    62→    62→      case 'w': return { top: '50%', left: offset, transform: 'translateY(-50%)' };
    63→    63→      default: return {};
    64→    64→    }
    65→    65→  };
    66→    66→
    67→    67→  // Accessible label for screen readers + arrow-key nudging for keyboard users.
    68→    68→  const ariaLabel = `${position} resize handle`;
    69→    69→  const handleKeyDown = (e: React.KeyboardEvent) => {
    70→    70→    if (e.key === 'Enter' || e.key === ' ') {
    71→    71→      e.preventDefault();
    72→    72→      // Keyboard users can't drag, but Enter/Space focuses the element for
    73→    73→      // subsequent arrow-key nudging handled by the canvas keyboard layer.
    74→    74→    }
    75→    75→  };
    76→    76→
    77→    77→  return (
    78→    78→    <div
    79→    79→      data-resize-handle
    80→    80→      role="button"
    81→    81→      tabIndex={0}
    82→    82→      aria-label={ariaLabel}
    83→    83→      className={`absolute ${size} bg-white border-2 border-primary rounded-sm hover:bg-primary hover:border-primary transition-colors focus:outline-none focus:ring-2 focus:ring-primary/60`}
    84→    84→      style={{ ...getPositionStyle() as React.CSSProperties, cursor, zIndex: 10000, pointerEvents: 'auto' }}
    85→    85→      onMouseDown={(e) => {
    86→    86→        if (e.button === 1) {
    87→    87→          e.preventDefault();
    88→    88→          useDesignerStore.getState().startPanning(e.clientX, e.clientY);
    89→    89→          return;
    90→    90→        }
    91→    91→        if (e.button !== 0) return;
    92→    92→        e.stopPropagation();
    93→    93→        onResizeStart(position, e);
    94→    94→      }}
    95→    95→      onKeyDown={handleKeyDown}
    96→    96→    />
    97→    97→  );
    98→    98→}

   151→   151→// Rectangle corner handle — appears at each of the 4 corner points of a
   152→   152→// selected rectangle. Behavior is ALT-gated:
   153→   153→//  - Alt+drag  → individual corner deformation (the corner moves independently,
   154→   154→//    the other 3 stay fixed). On a fresh axis-aligned rect this lazily
   155→   155→//    converts it to a polygon; on an already-deformed rect it just moves the
   156→   156→//    dragged corner.
   157→   157→//  - Plain drag (no Alt) → standard corner resize (nw/ne/se/sw). For a normal
   158→   158→//    rect this preserves the axis-aligned box; for a deformed rect the polygon
   159→   159→//    resizes per-edge (the opposite corner stays fixed).
   160→   160→// Alignment modifiers (held during an Alt+drag). The DRAGGED corner is the one
   161→   161→// that moves into alignment; the partner corner(s) stay fixed:
   162→   162→//  - Alt+Shift             → horizontal align (dragged corner's Y snaps to its
   163→   163→//                            horizontal partner's Y).
   164→   164→//  - Alt+Ctrl              → vertical align (dragged corner's X snaps to its
   165→   165→//                            vertical partner's X).
   166→   166→//  - Alt+Shift+Ctrl        → both (dragged corner snaps to (vPartner.X, hPartner.Y)).
   167→   167→function CornerHandle({
   168→   168→  x,
   169→   169→  y,
   170→   170→  cornerIndex,
   171→   171→  isCustom,
   172→   172→  rotation,
   173→   173→  onCornerMouseDown,
   174→   174→}: {
   175→   175→  x: number;
   176→   176→  y: number;
   177→   177→  cornerIndex: number; // 0=TL, 1=TR, 2=BR, 3=BL
   178→   178→  isCustom: boolean; // whether the rectangle already has deformed corners
   179→   179→  rotation: number; // rectangle rotation in degrees (for rotation-aware cursor)
   180→   180→  onCornerMouseDown: (cornerIndex: number, altHeld: boolean, e: React.MouseEvent) => void;
   181→   181→}) {
   182→   182→  // Rotation-aware cursor: maps corner index to its handle position (nw/ne/se/sw)
   183→   183→  // and generates a cursor rotated to the visual screen direction. At 0° this
   184→   184→  // returns the native nwse/nesw built-in; at other angles a custom SVG arrow.
   185→   185→  const handlePos = CORNER_HANDLE_MAP[cornerIndex];
   186→   186→  const cursor = getResizeCursor(handlePos, rotation);
   187→   187→  return (
   188→   188→    <div
   189→   189→      data-resize-handle
   190→   190→      data-corner-handle
   191→   191→      className={`absolute w-2.5 h-2.5 rounded-sm transition-colors ${
   192→   192→        isCustom
   193→   193→          ? 'bg-white border-2 border-purple-500 hover:bg-purple-500 hover:border-purple-500'
   194→   194→          : 'bg-white border-2 border-primary hover:bg-primary hover:border-primary'
   195→   195→      }`}
   196→   196→      style={{
   197→   197→        left: x - 5,
   198→   198→        top: y - 5,
   199→   199→        cursor,
   200→   200→        zIndex: 10000,
   201→   201→        pointerEvents: 'auto',
   202→   202→      }}
   203→   203→      title={isCustom ? 'Alt+drag to move corner · Alt+Shift horizontal align · Alt+Ctrl vertical align · Alt+Shift+Ctrl both' : 'Alt+drag to move corner independently'}
   204→   204→      onMouseDown={(e) => {
   205→   205→        if (e.button === 1) {
   206→   206→          e.preventDefault();
   207→   207→          useDesignerStore.getState().startPanning(e.clientX, e.clientY);
   208→   208→          return;
   209→   209→        }
   210→   210→        if (e.button !== 0) return;
   211→   211→        e.stopPropagation();
   212→   212→        onCornerMouseDown(cornerIndex, e.altKey, e);
   213→   213→      }}
   214→   214→    />
   215→   215→  );
   216→   216→}

   426→   426→  // units. With `fr`, the browser redistributes all track sizes whenever any
   427→   427→  // fraction changes, and sub-pixel rounding can cause other rows/columns to
   428→   428→  // shift by ~1px during a resize drag. With explicit `px` values, each
   429→   429→  // track's size is computed independently and stays stable when other tracks
   430→   430→  // change — only the two resizing tracks actually change size. The last
   431→   431→  // track absorbs any rounding remainder so the grid fills the element exactly.
   432→   432→  const cw = ensureColWidths(props.cols, props.colWidths);
   433→   433→  const rh = ensureRowHeights(props.rows, props.rowHeights);
   434→   434→  const totalColFr = cw.reduce((a, b) => a + b, 0) || 1;
   435→   435→  const visibleRowHeights = props.showHeader ? rh : rh.slice(1);
   436→   436→  const totalRowFr = (visibleRowHeights.length > 0 ? visibleRowHeights : [1]).reduce((a, b) => a + b, 0) || 1;
   437→   437→
   438→   438→  // Compute explicit pixel sizes; the last track gets the remainder to avoid
   439→   439→  // gaps/overflow from rounding.
   440→   440→  const computeTrackSizes = (fracs: number[], total: number, totalFr: number): string[] => {
   441→   441→    if (fracs.length === 0) return ['100%'];
   442→   442→    const sizes = fracs.map(f => (f / totalFr) * total);
   443→   443→    // Last track absorbs the rounding difference
   444→   444→    const allocated = sizes.slice(0, -1).reduce((a, b) => a + b, 0);
   445→   445→    sizes[sizes.length - 1] = total - allocated;
   446→   446→    return sizes.map(s => `${s}px`);
   447→   447→  };
   448→   448→
   449→   449→  const gridTemplateColumns = computeTrackSizes(cw, element.width, totalColFr).join(' ');
   450→   450→  const gridTemplateRows = computeTrackSizes(
   451→   451→    visibleRowHeights.length > 0 ? visibleRowHeights : [1],
   452→   452→    element.height,
   453→   453→    totalRowFr
   454→   454→  ).join(' ');
   455→   455→
   456→   456→  // Build a Set of selected cell keys for quick lookup.
   457→   457→  // IMPORTANT: only include cells when THIS table is the owner of the
   458→   458→  // selection (tableCellSelectTableId === element.id). Without this guard,
   459→   459→  // every table on the canvas would highlight cells at the same {row, col}
   460→   460→  // as the selected ones, making it look like all tables got selected.
   461→   461→  const selectedCellSet = useMemo(() => {
   462→   462→    const set = new Set<string>();
   463→   463→    if (!selectionOwnsTable) return set;
   464→   464→    for (const cell of selectedTableCells) {
   465→   465→      set.add(`${cell.row}-${cell.col}`);
   466→   466→    }
   467→   467→    return set;
   468→   468→  }, [selectedTableCells, selectionOwnsTable]);
   469→   469→
   470→   470→  // Render all visible cells as grid children
   471→   471→  const renderCells = () => {
   472→   472→    const cells: React.ReactNode[] = [];
   473→   473→
   474→   474→    for (let r = 0; r < props.rows; r++) {
   475→   475→      // Skip header row if hidden
   476→   476→      if (r === 0 && !props.showHeader) continue;
   477→   477→
   478→   478→      const gridRow = props.showHeader ? r + 1 : r; // 1-based grid row
   479→   479→
   480→   480→      for (let c = 0; c < props.cols; c++) {
   481→   481→        const padding = getCellPadding(r, c, props);
   482→   482→        const bgColor = getCellBg(r, c, props);
   483→   483→        const textColor = getCellColor(r, c, props);
   484→   484→        const fontFamily = getCellFontFamily(r, c, props);
   485→   485→        const fontWeight = getCellFontWeight(r, c, props);
   486→   486→        const fontSize = getCellFontSize(r, c, props);
   487→   487→        const textAlign = getCellTextAlign(r, c, props);
   488→   488→        const verticalAlign = getCellVerticalAlign(r, c, props);
   489→   489→        const textTransform = getCellTextTransform(r, c, props);
   490→   490→        const rotation = getCellRotation(r, c, props);
   491→   491→
   492→   492→        // ── Alt-key cell selection visual feedback ──────────────────────────
   493→   493→        const cellKey = `${r}-${c}`;
   494→   494→        const isSelected = selectedCellSet.has(cellKey);
   495→   495→        const isHovered = altHeld && hoveredTableCell?.row === r && hoveredTableCell?.col === c && hoveredTableCell?.tableId === element.id;
   496→   496→
   497→   497→        // Collapsed border model:
   498→   498→        // - Every cell draws top & left borders (handles outer + inner edges)
   499→   499→        // - Only last row draws bottom border (outer bottom edge)
   500→   500→        // - Only last column draws right border (outer right edge)
   501→   501→        // This avoids double-thick inner borders since each edge is drawn exactly once.
   502→   502→        // For selected cells, borders stay normal — only the dim overlay marks selection.
   503→   503→        const borderTop = getCellBorderTop(r, c, props) ? borderStr : 'none';
   504→   504→        const borderLeft = getCellBorderLeft(r, c, props) ? borderStr : 'none';
   505→   505→        const borderBottom = (r === props.rows - 1 && getCellBorderBottom(r, c, props)) ? borderStr : 'none';
   506→   506→        const borderRight = (c === props.cols - 1 && getCellBorderRight(r, c, props)) ? borderStr : 'none';
   507→   507→
   508→   508→        // Corner radius: apply to corner cells so borders follow the curve
   509→   509→        const isFirstVisibleRow = props.showHeader ? r === 0 : r === 1;
   510→   510→        const isLastRow = r === props.rows - 1;
   511→   511→        const isFirstCol = c === 0;
   512→   512→        const isLastCol = c === props.cols - 1;
   513→   513→        const cellBorderRadius = `${isFirstVisibleRow && isFirstCol ? radii.topLeft : 0}px ${isFirstVisibleRow && isLastCol ? radii.topRight : 0}px ${isLastRow && isLastCol ? radii.bottomRight : 0}px ${isLastRow && isFirstCol ? radii.bottomLeft : 0}px`;
   514→   514→
   515→   515→        // Vertical alignment via flexbox
   516→   516→        const justifyContent = verticalAlign === 'top'
   517→   517→          ? 'flex-start'
   518→   518→          : verticalAlign === 'bottom'
   519→   519→            ? 'flex-end'
   520→   520→            : 'center';
   521→   521→
   522→   522→        // For 'capitalize', we apply proper title case via JS (first letter uppercase, rest lowercase)
   523→   523→        // because CSS text-transform: capitalize only uppercases first letters without lowercasing the rest.
   524→   524→        const effectiveTextTransform = textTransform === 'capitalize' ? 'none' : textTransform;
   525→   525→        const cellText = props.cellData[r]?.[c] || '';
   526→   526→        const displayText = textTransform === 'capitalize' ? properCapitalize(cellText) : cellText;
   527→   527→
   528→   528→        cells.push(
   529→   529→          <div
   530→   530→            key={`${r}-${c}`}
   531→   531→            className="overflow-hidden"
   532→   532→            style={{
   533→   533→              gridRow,
   534→   534→              gridColumn: c + 1,
   535→   535→              padding: `${padding}px`,
   536→   536→              backgroundColor: bgColor,
   537→   537→              color: textColor,
   538→   538→              fontFamily,
   539→   539→              fontWeight,
   540→   540→              fontSize: `${fontSize}px`,
   541→   541→              textAlign: textAlign as 'left' | 'center' | 'right' | 'justify',
   542→   542→              ...(textAlign === 'justify' ? { textAlignLast: 'left' as const, textJustify: 'inter-character' as const, hyphens: 'auto' as const } : {}),
   543→   543→              textTransform: effectiveTextTransform as 'none' | 'uppercase' | 'lowercase',
   544→   544→              borderTop,
   545→   545→              borderBottom,
   546→   546→              borderLeft,
   547→   547→              borderRight,
   548→   548→              borderRadius: cellBorderRadius,
   549→   549→              boxSizing: 'border-box',
   550→   550→              // Use flexbox for vertical alignment, prevent content from expanding the cell
   551→   551→              display: 'flex',
   552→   552→              alignItems: justifyContent,
   553→   553→              minHeight: 0, // critical: allows grid to shrink below content size
   554→   554→              minWidth: 0,
   555→   555→              ...(rotation ? { transform: `rotate(${rotation}deg)` } : {}),
   556→   556→              position: 'relative',
   557→   557→              cursor: altHeld && isTableSelected ? 'pointer' : undefined,
   558→   558→            }}
   559→   559→            onDoubleClick={() => handleCellDoubleClick(r, c)}
   560→   560→            onMouseEnter={() => handleCellMouseEnter(r, c)}
   561→   561→            onMouseLeave={handleCellMouseLeave}
   562→   562→            onContextMenu={(e) => handleCellContextMenu(e, r, c)}
   563→   563→            onPointerDown={(e) => handleCellPointerDown(e, r, c)}
   564→   564→            onPointerUp={handleCellPointerUp}
   565→   565→          >
   566→   566→            {/* Alt-hover dim overlay — only when the table itself is selected */}
   567→   567→            {isHovered && !isSelected && isTableSelected && (
   568→   568→              <div
   569→   569→                className="absolute inset-0 pointer-events-none"
   570→   570→                style={{ backgroundColor: 'rgba(0, 0, 0, 0.08)', zIndex: 1 }}
   571→   571→              />
   572→   572→            )}
   573→   573→            {/* Selected cell dim overlay */}
   574→   574→            {isSelected && (
   575→   575→              <div
   576→   576→                className="absolute inset-0 pointer-events-none"
   577→   577→                style={{ backgroundColor: 'rgba(0, 0, 0, 0.06)', zIndex: 1 }}
   578→   578→              />
   579→   579→            )}
   580→   580→            {editingCell?.r === r && editingCell?.c === c ? (
   581→   581→              <input
   582→   582→                autoFocus
   583→   583→                value={editValue}
   584→   584→                onChange={(e) => setEditValue(e.target.value)}
   585→   585→                onBlur={handleCellBlur}
   586→   586→                onKeyDown={(e) => e.key === 'Enter' && handleCellBlur()}
   587→   587→                className="w-full bg-transparent outline-none border-none p-0 m-0"
   588→   588→                style={{
   589→   589→                  fontSize: 'inherit',
   590→   590→                  fontWeight: 'inherit',
   591→   591→                  color: 'inherit',
   592→   592→                  textAlign: 'inherit',
   593→   593→                  lineHeight: 1.2,
   594→   594→                }}
   595→   595→              />
   596→   596→            ) : (
   597→   597→              <span
   598→   598→                style={{
   599→   599→                  display: 'block',
   600→   600→                  width: '100%',
   601→   601→                  overflow: 'hidden',
   602→   602→                  textOverflow: 'ellipsis',
   603→   603→                  whiteSpace: 'nowrap',
   604→   604→                  lineHeight: 1.2,
   605→   605→                  position: 'relative',
   606→   606→                  zIndex: 2,
   607→   607→                }}
   608→   608→              >
   609→   609→                {displayText}
   610→   610→              </span>
   611→   611→            )}
   612→   612→          </div>
   613→   613→        );
   614→   614→      }
   615→   615→    }
   616→   616→
   617→   617→    return cells;
   618→   618→  };
   619→   619→
   620→   620→  const radii = resolveCornerRadius(props.cornerRadius);
   621→   621→
   622→   622→  return (
   623→   623→    <div
   624→   624→      className="w-full h-full"
   625→   625→      style={{
   626→   626→        opacity: props.opacity,
   627→   627→        borderRadius: `${radii.topLeft}px ${radii.topRight}px ${radii.bottomRight}px ${radii.bottomLeft}px`,
   628→   628→      }}
   629→   629→    >
   630→   630→      <div
   631→   631→        style={{
   632→   632→          display: 'grid',
   633→   633→          gridTemplateRows,
   634→   634→          gridTemplateColumns,
   635→   635→          width: '100%',
   636→   636→          height: '100%',
   637→   637→          fontSize: `${props.fontSize}px`,
   638→   638→          overflow: 'hidden',
   639→   639→          position: 'relative',
   640→   640→          userSelect: altHeld && isTableSelected ? 'none' : undefined,
   641→   641→        }}
   642→   642→      >
   643→   643→        {renderCells()}
   644→   644→      </div>
   645→   645→    </div>
   646→   646→  );
   647→   647→});
