    62→// ---- Debounced update hook ----
    63→// Debounces rapid numeric input changes so the store isn't updated on every keystroke.
    64→function useDebouncedCallback<T extends (...args: never[]) => void>(
    65→  callback: T,
    66→  delay: number,
    67→): T {
    68→  const timeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    69→  const callbackRef = useRef(callback);
    70→  // Update the callback ref in an effect to avoid accessing ref during render
    71→  useEffect(() => {
    72→    callbackRef.current = callback;
    73→  });
    74→  // Cleanup on unmount
    75→  useEffect(() => {
    76→    return () => {
    77→      if (timeoutRef.current) clearTimeout(timeoutRef.current);
    78→    };
    79→  }, []);
    80→
    81→  return useCallback(
    82→    (...args: Parameters<T>) => {
    83→      if (timeoutRef.current) clearTimeout(timeoutRef.current);
    84→      timeoutRef.current = setTimeout(() => callbackRef.current(...args), delay);
    85→    },
    86→    [delay],
    87→  ) as T;
    88→}

    92→function remapOverridesForRowSwap(
    93→  overrides: Record<string, CellOverride>,
    94→  rowA: number,
    95→  rowB: number
    96→): Record<string, CellOverride> {
    97→  const result: Record<string, CellOverride> = {};
    98→  for (const [key, val] of Object.entries(overrides)) {
    99→    const [r, c] = key.split('-').map(Number);
   100→    let newR = r;
   101→    if (r === rowA) newR = rowB;
   102→    else if (r === rowB) newR = rowA;
   103→    result[`${newR}-${c}`] = val;
   104→  }
   105→  return result;
   106→}

   156→// Remap overrides when a row/col is moved from `fromIndex` to `toIndex` (arbitrary distance)
   157→function remapOverridesForInsertion(
   158→  overrides: Record<string, CellOverride>,
   159→  fromIndex: number,
   160→  toIndex: number,
   161→  axis: 'row' | 'col'
   162→): Record<string, CellOverride> {
   163→  if (fromIndex === toIndex) return overrides;
   164→  const result: Record<string, CellOverride> = {};
   165→  for (const [key, val] of Object.entries(overrides)) {
   166→    const [r, c] = key.split('-').map(Number);
   167→    let newIdx = axis === 'row' ? r : c;
   168→    if (newIdx === fromIndex) {
   169→      newIdx = toIndex;
   170→    } else if (fromIndex < toIndex) {
   171→      // Moving down/right: shift items between from+1 and to up/left by 1
   172→      if (newIdx > fromIndex && newIdx <= toIndex) newIdx -= 1;
   173→    } else {
   174→      // Moving up/left: shift items between to and from-1 down/right by 1
   175→      if (newIdx >= toIndex && newIdx < fromIndex) newIdx += 1;
   176→    }
   177→    const newKey = axis === 'row' ? `${newIdx}-${c}` : `${r}-${newIdx}`;
   178→    result[newKey] = val;
   179→  }
   180→  return result;
   181→}

   274→// ─── Accordion Section ───────────────────────────────────────────────────
   275→// Replaces SectionTitle with a collapsible card wrapper.
   276→// - Collapsible with smooth animation (Suggestion 1)
   277→// - Visual card container (Suggestion 6)
   278→// - Auto-expands when search term matches the label (Suggestion 7)
   279→// - `defaultOpen` controls initial state; rarely-used sections default to closed
   280→// - `keywords` allows search to match additional terms beyond the label
   281→function AccordionSection({
   282→  label,
   283→  defaultOpen = false,
   284→  forceOpen = false,
   285→  keywords,
   286→  children,
   287→}: {
   288→  label: string;
   289→  defaultOpen?: boolean;
   290→  forceOpen?: boolean;
   291→  keywords?: string;
   292→  children: React.ReactNode;
   293→}) {
   294→  const [isOpen, setIsOpen] = useState(defaultOpen);
   295→  const searchTerm = useSearchTerm();
   296→
   297→  // Auto-expand when search matches the label or keywords
   298→  const searchText = [label, keywords].filter(Boolean).join(' ').toLowerCase();
   299→  const matchesSearch = searchTerm
   300→    ? searchText.includes(searchTerm.toLowerCase())
   301→    : false;
   302→
   303→  // If search is active and matches, force open; if search is active but doesn't match, force closed
   304→  // If forceOpen is true, always keep open
   305→  const effectiveOpen = searchTerm ? matchesSearch : (forceOpen || isOpen);
   306→
   307→  return (
   308→    <div
   309→      className="rounded-lg border border-border/60 bg-muted/20 overflow-hidden"
   310→      data-section-label={label.toLowerCase()}
   311→    >
   312→      <button
   313→        type="button"
   314→        className="w-full flex items-center gap-1.5 px-2.5 py-1.5 text-left hover:bg-muted/40 transition-colors duration-150"
   315→        onClick={() => setIsOpen(!isOpen)}
   316→      >
   317→        <ChevronRightIcon
   318→          className={`h-3 w-3 text-muted-foreground/60 shrink-0 transition-transform duration-200 ${
   319→            effectiveOpen ? 'rotate-90' : ''
   320→          }`}
   321→        />
   322→        <span className="text-[10px] font-semibold text-muted-foreground uppercase tracking-wider">
   323→          {label}
   324→        </span>
   325→      </button>
   326→      <div
   327→        className="grid transition-[grid-template-rows] duration-250 ease-[cubic-bezier(0.33,1,0.68,1)]"
   328→        style={{ gridTemplateRows: effectiveOpen ? '1fr' : '0fr' }}
   329→      >
   330→        <div className="overflow-hidden">
   331→          <div className="px-2.5 pb-2.5 pt-0.5 space-y-2">
   332→            {children}
   333→          </div>
   334→        </div>
   335→      </div>
   336→    </div>
   337→  );
   338→}

   502→// Text Properties — base/default formatting for text elements
   503→// Note: Per-character formatting is now available via the canvas text editor
   504→// (double-click a text element). These properties serve as the element-wide defaults.
   505→function TextPropertiesPanel({ element }: { element: CanvasElement }) {
   506→  const props = (element.properties as { type: 'text'; data: TextProperties }).data;
   507→  const updateElementProperties = useDesignerStore((s) => s.updateElementProperties);
   508→  const editingTextElementId = useDesignerStore((s) => s.editingTextElementId);
   509→  const activeTextEditor = useDesignerStore((s) => s.activeTextEditor);
   510→
   511→  const isEditing = editingTextElementId === element.id;
   512→
   513→  const update = useCallback((updates: Partial<TextProperties>) => {
   514→    let finalUpdates = updates;
   515→    // When textAlign is changed, apply it to the entire content
   516→    if (updates.textAlign) {
   517→      if (isEditing && activeTextEditor) {
   518→        // While editing: use TipTap's setTextAlign to update all paragraphs
   519→        activeTextEditor.chain().focus().setTextAlign(updates.textAlign).run();
   520→      } else {
   521→        // While not editing: update the HTML content directly to change/add
   522→        // text-align on all block-level elements (p, h1-h6, li, etc.)
   523→        const align = updates.textAlign;
   524→        let updatedContent = props.content;
   525→        // Step 1: For tags with style="..." — replace or add text-align
   526→        updatedContent = updatedContent.replace(
   527→          /(<(?:p|h[1-6]|li|div|blockquote)[^>]*?style=")([^"]*)(")/gi,
   528→          (_match, styleStart: string, styleContent: string, styleEnd: string) => {
   529→            if (styleContent.includes('text-align')) {
   530→              // Replace existing text-align value
   531→              return `${styleStart}${styleContent.replace(/text-align:\s*[^;"]+/gi, `text-align: ${align}`)}${styleEnd}`;
   532→            }
   533→            // Add text-align to existing style
   534→            const sep = styleContent && !styleContent.endsWith(';') ? ';' : '';
   535→            return `${styleStart}${styleContent}${sep}text-align: ${align}${styleEnd}`;
   536→          }
   537→        );
   538→        // Step 2: For tags without any style attribute — add style="text-align: X"
   539→        updatedContent = updatedContent.replace(
   540→          /<(p|h[1-6]|li|div|blockquote)(\s[^>]*?)?>/gi,
   541→          (match, tag: string, attrs: string | undefined) => {
   542→            if (attrs && attrs.includes('style=')) return match; // already handled in step 1
   543→            return `<${tag}${attrs || ''} style="text-align: ${align}">`;
   544→          }
   545→        );
   546→        finalUpdates = { ...updates, content: updatedContent };
   547→      }
   548→    }
   549→    updateElementProperties(element.id, {
   550→      type: 'text',
   551→      data: { ...props, ...finalUpdates },
   552→    });
   553→  }, [element.id, props, updateElementProperties, isEditing, activeTextEditor]);
   554→
   555→  return (
