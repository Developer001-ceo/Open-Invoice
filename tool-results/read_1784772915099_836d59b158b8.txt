    22→export function FloatingCard({ card, element, multiSelectIds, onBringToFront, boundsRef }: FloatingCardProps) {
    23→  const removeFloatingCard = useDesignerStore((s) => s.removeFloatingCard);
    24→  const clearCardInitialPosition = useDesignerStore((s) => s.clearCardInitialPosition);
    25→  const spotlightOpen = useDesignerStore((s) => s.spotlightOpen);
    26→  const settingsOpen = useDesignerStore((s) => s.settingsOpen);
    27→  // Subscribe to ALL modal states so floating cards hide behind any modal
    28→  const newProjectOpen = useDesignerStore((s) => s.newProjectOpen);
    29→  const exportPdfOpen = useDesignerStore((s) => s.exportPdfOpen);
    30→  const largePreviewOpen = useDesignerStore((s) => s.largePreviewOpen);
    31→  const welcomeModalOpen = useDesignerStore((s) => s.welcomeModalOpen);
    32→  const predefinedBlocksOpen = useDesignerStore((s) => s.predefinedBlocksOpen);
    33→  // Any modal open → cards go behind (z-40, below the z-50 dialogs)
    34→  const anyModalOpen = settingsOpen || newProjectOpen || exportPdfOpen || largePreviewOpen || welcomeModalOpen || predefinedBlocksOpen;
    35→
    36→  // Fly-in animation: start at initialX/initialY (spotlight card position) if provided,
    37→  // then animate to the target position (card.x/card.y)
    38→  const hasInitialPosition = card.initialX !== undefined && card.initialY !== undefined;
    39→  const [position, setPosition] = useState({
    40→    x: hasInitialPosition ? card.initialX! : card.x,
    41→    y: hasInitialPosition ? card.initialY! : card.y,
    42→  });
    43→  const [isAnimating, setIsAnimating] = useState(hasInitialPosition);
    44→  const [isMinimized, setIsMinimized] = useState(card.minimized);
    45→  const [isDragging, setIsDragging] = useState(false);
    46→  const [dragOffset, setDragOffset] = useState({ x: 0, y: 0 });
    47→  const cardRef = useRef<HTMLDivElement>(null);
    48→
    49→  // Fly-in animation: after mount, animate from initial position to target position
    50→  useEffect(() => {
    51→    if (!hasInitialPosition) return;
    52→    let timer: ReturnType<typeof setTimeout> | null = null;
    53→    let raf2: number | null = null;
    54→    // Use double-rAF to ensure the browser has painted the initial position first
    55→    const raf1 = requestAnimationFrame(() => {
    56→      raf2 = requestAnimationFrame(() => {
    57→        setPosition({ x: card.x, y: card.y });
    58→        // Clear animation state after transition completes
    59→        timer = setTimeout(() => {
    60→          setIsAnimating(false);
    61→          clearCardInitialPosition(card.id);
    62→        }, 450); // Match the CSS transition duration
    63→      });
    64→    });
    65→    return () => {
    66→      cancelAnimationFrame(raf1);
    67→      if (raf2 !== null) cancelAnimationFrame(raf2);
    68→      if (timer !== null) clearTimeout(timer);
    69→    };
    70→  }, []);
    71→
    72→  // Handle dragging — RAF-throttled position updates to avoid layout thrashing
    73→  const rafRef = useRef<number | null>(null);
    74→  const pendingPosRef = useRef<{ x: number; y: number } | null>(null);
    75→
    76→  const handleMouseDown = useCallback((e: React.MouseEvent) => {
    77→    if ((e.target as HTMLElement).closest('[data-no-drag]')) return;
    78→    e.preventDefault();
    79→    e.stopPropagation();
    80→    setIsDragging(true);
    81→    setDragOffset({
    82→      x: e.clientX - position.x,
    83→      y: e.clientY - position.y,
    84→    });
    85→    onBringToFront(card.id);
    86→  }, [position, onBringToFront, card.id]);
    87→
    88→  useEffect(() => {
    89→    if (!isDragging) return;
    90→
    91→    const handleMouseMove = (e: MouseEvent) => {
    92→      let newX = e.clientX - dragOffset.x;
    93→      let newY = e.clientY - dragOffset.y;
    94→
    95→      // Constrain to bounds if provided
    96→      if (boundsRef?.current) {
    97→        const rect = boundsRef.current.getBoundingClientRect();
    98→        newX = Math.max(rect.left, Math.min(newX, rect.right - getCardWidthPx(card.sectionLabel)));
    99→        newY = Math.max(rect.top, Math.min(newY, rect.bottom - 40));
   100→      }
   101→
   102→      // Throttle position updates via RAF
   103→      pendingPosRef.current = { x: newX, y: newY };
   104→      if (rafRef.current === null) {
   105→        rafRef.current = requestAnimationFrame(() => {
   106→          if (pendingPosRef.current) {
   107→            setPosition(pendingPosRef.current);
   108→            pendingPosRef.current = null;
   109→          }
   110→          rafRef.current = null;
   111→        });
   112→      }
   113→    };
   114→
   115→    const handleMouseUp = () => {
   116→      // Flush any pending position before stopping drag
   117→      if (pendingPosRef.current) {
   118→        setPosition(pendingPosRef.current);
   119→        pendingPosRef.current = null;
   120→      }
   121→      if (rafRef.current !== null) {
   122→        cancelAnimationFrame(rafRef.current);
   123→        rafRef.current = null;
   124→      }
   125→      setIsDragging(false);
   126→    };
   127→
   128→    window.addEventListener('mousemove', handleMouseMove);
   129→    window.addEventListener('mouseup', handleMouseUp);
   130→    return () => {
   131→      window.removeEventListener('mousemove', handleMouseMove);
   132→      window.removeEventListener('mouseup', handleMouseUp);
   133→    };
   134→  }, [isDragging, dragOffset, boundsRef]);
   135→
   136→  const handleClose = useCallback(() => {
   137→    removeFloatingCard(card.id);
   138→  }, [removeFloatingCard, card.id]);
   139→
   140→  const handleToggleMinimize = useCallback(() => {
   141→    setIsMinimized(!isMinimized);
   142→  }, [isMinimized]);
   143→
   144→  // Render the appropriate property content based on sectionLabel
   145→  const renderContent = () => {
   146→    return <PropertyContent sectionLabel={card.sectionLabel} element={element} multiSelectIds={multiSelectIds} />;
   147→  };
   148→
   149→  return (
   150→    <div
   151→      ref={cardRef}
   152→      className="fixed select-none"
   153→      style={{
   154→        left: position.x,
   155→        top: position.y,
   156→        // When ANY modal is open (z-50), push floating cards behind it (z-40)
   157→        // When spotlight is open, force z-index below the spotlight overlay (z-99999)
   158→        // Otherwise use the card's z-index for stacking among floating cards
   159→        zIndex: anyModalOpen ? 40 : spotlightOpen ? 100 : card.zIndex,
   160→        // Smooth transition for fly-in animation (not during drag)
   161→        transition: isAnimating && !isDragging
   162→          ? 'left 400ms cubic-bezier(0.34, 1.56, 0.64, 1), top 400ms cubic-bezier(0.34, 1.56, 0.64, 1)'
   163→          : isAnimating
   164→          ? 'none'
   165→          : 'none',
   166→      }}
   167→      onMouseDown={() => onBringToFront(card.id)}
   168→    >
   169→      <div
   170→        className={`${getCardWidthClass(card.sectionLabel)} rounded-lg border border-border/60 bg-card shadow-xl overflow-hidden transition-all duration-200 relative ${
   171→          isDragging ? 'shadow-2xl cursor-grabbing' : 'cursor-grab'
   172→        } ${spotlightOpen || anyModalOpen ? 'pointer-events-none' : ''}`}
   173→        style={spotlightOpen || anyModalOpen ? { opacity: 0.4, filter: 'saturate(0.3) brightness(0.6)' } : undefined}
   174→      >
   175→        {/* Title bar */}
   176→        <div
   177→          className="flex items-center gap-1.5 px-2.5 py-1.5 bg-muted/30 border-b border-border/40 hover:bg-muted/50 transition-colors"
   178→          onMouseDown={handleMouseDown}
   179→          onDoubleClick={handleToggleMinimize}
   180→        >
   181→          <GripHorizontal className="h-3.5 w-3.5 text-muted-foreground/40 shrink-0" />
   182→          <span className="text-[11px] font-medium text-foreground truncate flex-1">
   183→            {card.sectionLabel}
   184→          </span>
   185→          <button
   186→            data-no-drag
   187→            type="button"
   188→            className="p-0.5 rounded hover:bg-muted transition-colors"
   189→            onClick={handleToggleMinimize}
   190→            title={isMinimized ? 'Expand' : 'Minimize'}
   191→          >
   192→            {isMinimized ? (
   193→              <ChevronDown className="h-3 w-3 text-muted-foreground" />
   194→            ) : (
   195→              <ChevronUp className="h-3 w-3 text-muted-foreground" />
   196→            )}
   197→          </button>
   198→          <button
   199→            data-no-drag
   200→            type="button"
   201→            className="p-0.5 rounded hover:bg-destructive/10 hover:text-destructive transition-colors"
   202→            onClick={handleClose}
   203→            title="Close"
   204→          >
   205→            <X className="h-3 w-3 text-muted-foreground hover:text-destructive" />
   206→          </button>
   207→        </div>
   208→
   209→        {/* Content */}
   210→        {!isMinimized && (
   211→          <div className="p-3 max-h-[70vh] overflow-y-auto">
   212→            {renderContent()}
   213→          </div>
   214→        )}
   215→
   216→        {/* Blue dim overlay when spotlight or settings is open */}
   217→        {(spotlightOpen || anyModalOpen) && (
   218→          <div className="absolute inset-0 bg-blue-500/20 pointer-events-none rounded-lg" />
   219→        )}
   220→      </div>
   221→    </div>
   222→  );
   223→}

   231→// ─── Dynamic card width based on section content ───────────────────────
   232→function getCardWidthClass(sectionLabel: string): string {
   233→  // Wide sections (320px) - complex UI with grids, color pickers, presets, etc.
   234→  const wideSections = new Set([
   235→    'Fill & Gradient', 'Gradient',
   236→    'Cell Properties',
   237→    'Shadow',
   238→    'Reflection',
   239→    'Text Content',
   240→    'Formatting',
   241→  ]);
   242→  // Medium-wide sections (280px) - multiple fields, color rows, checkbox grids
   243→  const mediumSections = new Set([
   244→    'Typography',
   245→    'Borders', 'Table Border',
   246→    'Corner Radius',
   247→    'Header Row',
   248→    'Cell Styling', 'Cell Defaults',
   249→    'Image Filters', 'Adjustments',
   250→    'Stroke',
   251→    'Glow',
   252→    'Reorder',
   253→  ]);
   254→
   255→  if (wideSections.has(sectionLabel)) return 'w-[320px]';
   256→  if (mediumSections.has(sectionLabel)) return 'w-[280px]';
   257→  return 'w-[260px]';
   258→}

   311→// ─── Cell Properties sub-component ─────────────────────────────────
   312→function CellPropertiesEditor({ element, props, update, compact }: { element: CanvasElement; props: TableProperties; update: (updates: Partial<TableProperties>) => void; compact?: boolean }) {
   313→  // Sync with store's Alt-key cell selection from the canvas.
   314→  // Only reflect the store selection when it belongs to THIS table — the
   315→  // store tracks tableCellSelectTableId so selections stay scoped per table
   316→  // (otherwise cells at the same {row, col} in other tables would light up).
   317→  const storeSelectedTableCells = useDesignerStore((s) => s.selectedTableCells);
   318→  const tableCellSelectTableId = useDesignerStore((s) => s.tableCellSelectTableId);
   319→  // Only reflect the store selection when it belongs to THIS table. Memoized
   320→  // so the dependency identity stays stable for useCallback/useMemo below.
   321→  const storeSelectionForThisTable = useMemo(
   322→    () => (tableCellSelectTableId === element.id ? storeSelectedTableCells : []),
   323→    [tableCellSelectTableId, element.id, storeSelectedTableCells]
   324→  );
   325→  const [localSelectedCells, setLocalSelectedCells] = useState<Set<string>>(new Set());
   326→
   327→  // Merge store's canvas cell selection with local selection.
   328→  // Store selection takes priority when present (from Alt+click on canvas).
   329→  const selectedCells = storeSelectionForThisTable.length > 0
   330→    ? new Set(storeSelectionForThisTable.map(cell => `${cell.row}-${cell.col}`))
   331→    : localSelectedCells;
   332→
   333→  // Wrapper for setSelectedCells that updates local state and clears store selection
   334→  const setSelectedCells = useCallback((updater: Set<string> | ((prev: Set<string>) => Set<string>)) => {
   335→    // Clear store selection when using the card mini-table to select
   336→    if (storeSelectionForThisTable.length > 0) {
   337→      useDesignerStore.getState().clearTableCellSelection();
   338→    }
   339→    setLocalSelectedCells((prev) => {
   340→      const resolved = typeof updater === 'function' ? updater(storeSelectionForThisTable.length > 0 ? new Set(storeSelectionForThisTable.map(cell => `${cell.row}-${cell.col}`)) : prev) : updater;
   341→      return resolved;
   342→    });
   343→  }, [storeSelectionForThisTable]);
   344→
   345→  const getCellOverride = (r: number, c: number): CellOverride => {
   346→    return props.cellOverrides[`${r}-${c}`] || {};
   347→  };
   348→
   349→  const updateCellOverride = (r: number, c: number, updates: Partial<CellOverride>) => {
   350→    const key = `${r}-${c}`;
   351→    const existing = props.cellOverrides[key] ?? {};
   352→    const newOverrides = { ...props.cellOverrides, [key]: { ...existing, ...updates } };
   353→    update({ cellOverrides: newOverrides });
   354→  };
   355→
   356→  const clearCellOverride = (r: number, c: number) => {
   357→    const key = `${r}-${c}`;
   358→    const newOverrides = { ...props.cellOverrides };
   359→    delete newOverrides[key];
   360→    update({ cellOverrides: newOverrides });
   361→  };
   362→
   363→  // ---- Multi-cell helpers ----
   364→
   365→  const toggleCellSelection = (r: number, c: number, multi: boolean) => {
   366→    const key = `${r}-${c}`;
   367→    setSelectedCells((prev) => {
   368→      const next = new Set(prev);
   369→      if (multi) {
   370→        if (next.has(key)) {
   371→          next.delete(key);
   372→        } else {
   373→          next.add(key);
   374→        }
   375→      } else {
   376→        if (next.size === 1 && next.has(key)) {
   377→          return prev;
   378→        }
   379→        next.clear();
   380→        next.add(key);
   381→      }
   382→      return next;
   383→    });
   384→  };
   385→
   386→  const parseCellKey = (key: string): { r: number; c: number } => {
   387→    const [r, c] = key.split('-').map(Number);
   388→    return { r, c };
   389→  };
   390→
   391→  // Update an override property for all selected cells at once
   392→  const updateSelectedCellsOverride = (overrideUpdate: Partial<CellOverride>) => {
   393→    const newOverrides = { ...props.cellOverrides };
   394→    for (const key of selectedCells) {
   395→      const existing = newOverrides[key] || {};
   396→      const merged = { ...existing, ...overrideUpdate };
   397→      const cleaned: CellOverride = {};
   398→      for (const [k, v] of Object.entries(merged)) {
   399→        if (v !== undefined) (cleaned as Record<string, unknown>)[k] = v;
   400→      }
   401→      if (Object.keys(cleaned).length === 0) {
   402→        delete newOverrides[key];
   403→      } else {
   404→        newOverrides[key] = cleaned;
   405→      }
   406→    }
   407→    update({ cellOverrides: newOverrides });
   408→  };
   409→
   410→  // Clear overrides for all selected cells
   411→  const clearSelectedCellsOverrides = () => {
   412→    const newOverrides = { ...props.cellOverrides };
   413→    for (const key of selectedCells) {
   414→      delete newOverrides[key];
   415→    }
   416→    update({ cellOverrides: newOverrides });
   417→  };
   418→
   419→  // Compute the common value of a property across all selected cells
   420→  const getMultiCellValue = <T,>(
   421→    getter: (r: number, c: number) => T,
   422→    fallback: T
   423→  ): { value: T; mixed: boolean } => {
   424→    if (selectedCells.size === 0) return { value: fallback, mixed: false };
   425→    const values: T[] = [];
   426→    for (const key of selectedCells) {
   427→      const { r, c } = parseCellKey(key);
   428→      values.push(getter(r, c));
   429→    }
   430→    const first = values[0];
   431→    const mixed = values.some((v) => v !== first);
   432→    return { value: mixed ? fallback : first, mixed };
   433→  };
   434→
   435→  // Computed override values for selected cells
   436→  const multiAlign = getMultiCellValue(
   437→    (r, c) => getCellOverride(r, c).textAlign ?? 'left',
   438→    'left'
   439→  );
   440→  const multiVAlign = getMultiCellValue(
   441→    (r, c) => getCellOverride(r, c).verticalAlign ?? 'middle',
   442→    'middle'
   443→  );
   444→  const multiPadding = getMultiCellValue(
   445→    (r, c) => getCellOverride(r, c).padding ?? props.cellPadding,
   446→    props.cellPadding
   447→  );
   448→  const multiBgColor = getMultiCellValue(
   449→    (r, c) => getCellOverride(r, c).bgColor || props.cellBg,
   450→    props.cellBg
   451→  );
   452→  const multiTextColor = getMultiCellValue(
   453→    (r, c) => getCellOverride(r, c).color || props.cellColor,
   454→    props.cellColor
   455→  );
   456→  const multiFontWeight = getMultiCellValue(
   457→    (r, c) => getCellOverride(r, c).fontWeight || 'normal',
   458→    'normal'
   459→  );
   460→  const multiFontSize = getMultiCellValue(
   461→    (r, c) => getCellOverride(r, c).fontSize || props.fontSize,
   462→    props.fontSize
   463→  );
   464→  const multiTextTransform = getMultiCellValue(
   465→    (r, c) => getCellOverride(r, c).textTransform || 'none',
   466→    'none'
   467→  );
   468→
   469→  // Linked cell border toggle — when toggling a side, also toggle the adjacent cell's opposite side
   470→  const toggleCellBorderLinked = (r: number, c: number, side: 'borderTop' | 'borderRight' | 'borderBottom' | 'borderLeft') => {
   471→    const current = props.cellOverrides[`${r}-${c}`] ?? {};
   472→    const currentVal = current[side] ?? true;
   473→    const newVal = !currentVal;
   474→
   475→    const newOverrides = { ...props.cellOverrides };
   476→
   477→    // Update the target cell
   478→    const targetKey = `${r}-${c}`;
   479→    const targetExisting = newOverrides[targetKey] || {};
   480→    const targetMerged = { ...targetExisting, [side]: newVal };
   481→    const targetCleaned: CellOverride = {};
   482→    for (const [k, v] of Object.entries(targetMerged)) {
   483→      if (v !== undefined) (targetCleaned as Record<string, unknown>)[k] = v;
   484→    }
   485→    if (Object.keys(targetCleaned).length === 0) {
   486→      delete newOverrides[targetKey];
   487→    } else {
   488→      newOverrides[targetKey] = targetCleaned;
   489→    }
   490→
   491→    // Determine adjacent cell and its opposite side
   492→    let adjR = r;
   493→    let adjC = c;
   494→    let adjSide: 'borderTop' | 'borderRight' | 'borderBottom' | 'borderLeft' = side;
   495→
   496→    switch (side) {
   497→      case 'borderTop':
   498→        adjR = r - 1;
   499→        adjSide = 'borderBottom';
   500→        break;
   501→      case 'borderBottom':
   502→        adjR = r + 1;
   503→        adjSide = 'borderTop';
   504→        break;
   505→      case 'borderLeft':
   506→        adjC = c - 1;
   507→        adjSide = 'borderRight';
   508→        break;
   509→      case 'borderRight':
   510→        adjC = c + 1;
   511→        adjSide = 'borderLeft';
   512→        break;
   513→    }
   514→
   515→    // Only update adjacent cell if it exists (not out of bounds)
   516→    if (adjR >= 0 && adjR < props.rows && adjC >= 0 && adjC < props.cols) {
   517→      const adjKey = `${adjR}-${adjC}`;
   518→      const adjExisting = newOverrides[adjKey] || {};

   522→        if (v !== undefined) (adjCleaned as Record<string, unknown>)[k] = v;
   523→      }
   524→      if (Object.keys(adjCleaned).length === 0) {
   525→        delete newOverrides[adjKey];
   526→      } else {
   527→        newOverrides[adjKey] = adjCleaned;
   528→      }
   529→    }

   531→    update({ cellOverrides: newOverrides });

   558→                          const newCellData = props.cellData.map((r2) => [...r2]);

   568→                        onClick={(e) => { e.stopPropagation(); toggleCellSelection(r, c, e.shiftKey || e.ctrlKey || e.metaKey); }}

   571→                    </td>
