   132→function PreviewTableElement({ element }: { element: CanvasElement }) {
   133→  const rawProps = (element.properties as { type: 'table'; data: TableProperties }).data;
   134→  const props = normalizeTableProps(rawProps);
   135→
   136→  const bw = props.borderWidth;
   137→  const borderStyleValue = props.borderStyle;
   138→  const borderColorValue = props.borderColor;
   139→  const borderStr = `${bw}px ${borderStyleValue} ${borderColorValue}`;
   140→
   141→  const visibleRows = props.showHeader ? props.rows : Math.max(props.rows - 1, 1);
   142→  const radii = resolveCornerRadius(props.cornerRadius);
   143→
   144→  // Convert colWidths/rowHeights proportional fractions into EXPLICIT pixel
   145→  // track sizes — matching the canvas rendering. We must NOT use CSS `fr`
   146→  // units here: `fr` tracks are expanded by min-content/max-content of cell
   147→  // content, so a wide cell would blow out the user's resized ratio and the
   148→  // preview would show default/equal sizes instead of the adjusted ones.
   149→  // Explicit px values (with the last track absorbing rounding) reproduce the
   150→  // canvas layout exactly.
   151→  const cw = ensureColWidths(props.cols, props.colWidths);
   152→  const rh = ensureRowHeights(props.rows, props.rowHeights);
   153→  const visibleRowHeights = props.showHeader ? rh : rh.slice(1);
   154→  const totalColFr = cw.reduce((a, b) => a + b, 0) || 1;
   155→  const visRh = visibleRowHeights.length > 0 ? visibleRowHeights : [1];
   156→  const totalRowFr = visRh.reduce((a, b) => a + b, 0) || 1;
   157→  const computeTrackSizes = (fracs: number[], total: number, totalFr: number): string[] => {
   158→    if (fracs.length === 0) return ['100%'];
   159→    const sizes = fracs.map((f) => (f / totalFr) * total);
   160→    const allocated = sizes.slice(0, -1).reduce((a, b) => a + b, 0);
   161→    sizes[sizes.length - 1] = total - allocated;
   162→    return sizes.map((s) => `${s}px`);
   163→  };
   164→  const gridTemplateColumns = computeTrackSizes(cw, element.width, totalColFr).join(' ');
   165→  const gridTemplateRows = computeTrackSizes(visRh, element.height, totalRowFr).join(' ');
   166→
   167→  const cells: React.ReactNode[] = [];
   168→  for (let r = 0; r < props.rows; r++) {
   169→    if (r === 0 && !props.showHeader) continue;
   170→    const gridRow = props.showHeader ? r + 1 : r;
   171→
   172→    for (let c = 0; c < props.cols; c++) {
   173→      const padding = getCellPadding(r, c, props);
   174→      const bgColor = getCellBg(r, c, props);
   175→      const textColor = getCellColor(r, c, props);
   176→      const fontWeight = getCellFontWeight(r, c, props);
   177→      const fontSize = getCellFontSize(r, c, props);
   178→      const textAlign = getCellTextAlign(r, c, props);
   179→      const verticalAlign = getCellVerticalAlign(r, c, props);
   180→
   181→      // Collapsed border model: each shared edge drawn once
   182→      const borderTop = getCellBorderTop(r, c, props) ? borderStr : 'none';
   183→      const borderLeft = getCellBorderLeft(r, c, props) ? borderStr : 'none';
   184→      const borderBottom = (r === props.rows - 1 && getCellBorderBottom(r, c, props)) ? borderStr : 'none';
   185→      const borderRight = (c === props.cols - 1 && getCellBorderRight(r, c, props)) ? borderStr : 'none';
   186→
   187→      // Corner radius: apply to corner cells so borders follow the curve
   188→      const isFirstVisibleRow = props.showHeader ? r === 0 : r === 1;
   189→      const isLastRow = r === props.rows - 1;
   190→      const isFirstCol = c === 0;
   191→      const isLastCol = c === props.cols - 1;
   192→      const cellBorderRadius = `${isFirstVisibleRow && isFirstCol ? radii.topLeft : 0}px ${isFirstVisibleRow && isLastCol ? radii.topRight : 0}px ${isLastRow && isLastCol ? radii.bottomRight : 0}px ${isLastRow && isFirstCol ? radii.bottomLeft : 0}px`;
   193→
   194→      const justifyContent = verticalAlign === 'top' ? 'flex-start' : verticalAlign === 'bottom' ? 'flex-end' : 'center';
   195→
   196→      cells.push(
   197→        <div
   198→          key={`${r}-${c}`}
   199→          style={{
   200→            gridRow,
   201→            gridColumn: c + 1,
   202→            overflow: 'hidden',
   203→            padding: `${padding}px`,
   204→            backgroundColor: bgColor,
   205→            color: textColor,
   206→            fontWeight,
   207→            fontSize: `${fontSize}px`,
   208→            textAlign: textAlign as 'left' | 'center' | 'right' | 'justify',
   209→            ...(textAlign === 'justify' ? { textAlignLast: 'left' as const, textJustify: 'inter-character' as const, hyphens: 'auto' as const } : {}),
   210→            borderTop,
   211→            borderBottom,
   212→            borderLeft,
   213→            borderRight,
   214→            borderRadius: cellBorderRadius,
   215→            boxSizing: 'border-box',
   216→            display: 'flex',
   217→            alignItems: justifyContent,
   218→            minHeight: 0,
   219→            minWidth: 0,
   220→          }}
   221→        >
   222→          <span
   223→            style={{
   224→              display: 'block',
   225→              width: '100%',
   226→              overflow: 'hidden',
   227→              textOverflow: 'ellipsis',
   228→              whiteSpace: 'nowrap',
   229→              lineHeight: 1.2,
   230→            }}
   231→          >
   232→            {props.cellData[r]?.[c] || ''}
   233→          </span>
   234→        </div>
   235→      );
   236→    }
   237→  }
   238→
   239→  return (
   240→    <div style={{
   241→      width: '100%',
   242→      height: '100%',
   243→      opacity: props.opacity,
   244→      borderRadius: `${radii.topLeft}px ${radii.topRight}px ${radii.bottomRight}px ${radii.bottomLeft}px`,
   245→    }}>
   246→      <div
   247→        style={{
   248→          display: 'grid',
   249→          gridTemplateRows,
   250→          gridTemplateColumns,
   251→          width: '100%',
   252→          height: '100%',
   253→          fontSize: `${props.fontSize}px`,
   254→          overflow: 'hidden',
   255→        }}
   256→      >
   257→        {cells}
   258→      </div>
   259→    </div>
   260→  );
   261→}

   263→function PreviewImageElement({ element }: { element: CanvasElement }) {
   264→  const props = (element.properties as { type: 'image'; data: ImageProperties }).data;
   265→  const effects: RectangleEffects = props.effects ?? { ...DEFAULT_EFFECTS };
   266→
   267→  // Compute combined box shadow (glow + shadow)
   268→  const combinedBoxShadow = (() => {
   269→    const glowBoxShadow = buildGlowCSS(effects.glow).boxShadow as string | undefined;
   270→    const shadowBoxShadow = buildShadowCSS(effects.shadow, props.borderRadius).boxShadow as string | undefined;
   271→    if (glowBoxShadow && shadowBoxShadow) return `${glowBoxShadow}, ${shadowBoxShadow}`;
   272→    if (glowBoxShadow) return glowBoxShadow;
   273→    if (shadowBoxShadow) return shadowBoxShadow;
   274→    return undefined;
   275→  })();
   276→
   277→  const reflectionMask = effects.reflection.enabled ? getReflectionMask(effects.reflection.preset) : undefined;
   278→
   279→  if (!props.src) {
   280→    return (
   281→      <div
   282→        style={{
   283→          width: '100%',
   284→          height: '100%',
   285→          display: 'flex',
   286→          flexDirection: 'column',
   287→          alignItems: 'center',
   288→          justifyContent: 'center',
   289→          backgroundColor: '#f9fafb',
   290→          border: '2px dashed #d1d5db',
   291→          borderRadius: `${props.borderRadius}px`,
   292→          opacity: props.opacity,
   293→          boxShadow: combinedBoxShadow,
   294→        }}
   295→      >
   296→        <span style={{ fontSize: '10px', color: '#9ca3af' }}>No image</span>
   297→      </div>
   298→    );
   299→  }
   300→
   301→  return (
   302→    <div style={{ position: 'relative', width: '100%', height: '100%' }}>
   303→      {/* Main image with glow + shadow */}
   304→      <div
   305→        style={{
   306→          width: '100%',
   307→          height: '100%',
   308→          overflow: 'hidden',
   309→          borderRadius: `${props.borderRadius}px`,
   310→          border: props.borderWidth > 0 ? `${props.borderWidth}px solid ${props.borderColor}` : 'none',
   311→          opacity: props.opacity,
   312→          boxShadow: combinedBoxShadow,
   313→          position: 'relative',
   314→          zIndex: 2,
   315→        }}
   316→      >
   317→        <img
   318→          src={props.src}
   319→          alt=""
   320→          style={{
   321→            width: '100%',
   322→            height: '100%',
   323→            objectFit: props.objectFit,
   324→            filter: `brightness(${props.brightness ?? 100}%) contrast(${props.contrast ?? 100}%) saturate(${props.saturation ?? 100}%) blur(${props.blur ?? 0}px)`,
   325→          }}
   326→          draggable={false}
   327→        />
   328→      </div>
   329→
   330→      {/* Reflection below */}
   331→      {effects.reflection.enabled && (
   332→        <div
   333→          style={{
   334→            position: 'absolute',
   335→            top: `calc(100% + ${effects.reflection.distance}px)`,
   336→            left: 0,
   337→            width: '100%',
   338→            height: `${effects.reflection.size}%`,
   339→            overflow: 'hidden',
   340→            zIndex: 1,
   341→            pointerEvents: 'none',
   342→            maskImage: reflectionMask,
   343→            WebkitMaskImage: reflectionMask,
   344→            borderRadius: `${props.borderRadius}px`,
   345→          }}
   346→        >
   347→          <div style={{
   348→            width: '100%',
   349→            height: '100%',
   350→            overflow: 'hidden',
   351→            borderRadius: `${props.borderRadius}px`,
   352→            border: props.borderWidth > 0 ? `${props.borderWidth}px solid ${props.borderColor}` : 'none',
   353→            transform: 'scaleY(-1)',
   354→            opacity: effects.reflection.opacity,
   355→          }}>
   356→            <img
   357→              src={props.src}
   358→              alt=""
   359→              style={{
   360→                width: '100%',
   361→                height: '100%',
   362→                objectFit: props.objectFit,
   363→                filter: `brightness(${props.brightness ?? 100}%) contrast(${props.contrast ?? 100}%) saturate(${props.saturation ?? 100}%) blur(${props.blur ?? 0}px)`,
   364→              }}
   365→              draggable={false}
   366→            />
   367→          </div>
   368→        </div>
   369→      )}
   370→    </div>
   371→  );
   372→}

   374→function PreviewLineElement({ element }: { element: CanvasElement }) {
   375→  const props = (element.properties as { type: 'line'; data: LineProperties }).data;
   376→
   377→  const startX = element.lineStartX ?? element.x;
   378→  const startY = element.lineStartY ?? (element.y + element.height / 2);
   379→  const endX = element.lineEndX ?? (element.x + element.width);
   380→  const endY = element.lineEndY ?? (element.y + element.height / 2);
   381→
   382→  const relStartX = startX - element.x;
   383→  const relStartY = startY - element.y;
   384→  const relEndX = endX - element.x;
   385→  const relEndY = endY - element.y;
   386→
   387→  const dashArray = props.strokeStyle === 'dashed' ? '8,4' : props.strokeStyle === 'dotted' ? '2,4' : 'none';
   388→
   389→  return (
   390→    <svg style={{ width: '100%', height: '100%', opacity: props.opacity, overflow: 'visible' }}>
   391→      <path
   392→        d={`M ${relStartX} ${relStartY} L ${relEndX} ${relEndY}`}
   393→        stroke={props.strokeColor}
   394→        strokeWidth={props.strokeWidth}
   395→        strokeDasharray={dashArray}
   396→        strokeLinecap={props.lineCap}
   397→        fill="none"
   398→      />
   399→    </svg>
   400→  );
   401→}

   403→function PreviewRectangleElement({ element }: { element: CanvasElement }) {
   404→  const props = (element.properties as { type: 'rectangle'; data: RectangleProperties }).data;
   405→  const effects: RectangleEffects = props.effects ?? { ...DEFAULT_EFFECTS };
   406→
   407→  // ── Custom-corner (deformed) rectangle: render as an SVG polygon ──────────
   408→  if (hasCustomCorners(element)) {
   409→    const corners = getRectCornerPoints(element);
   410→    // Build a rounded path so corner radius still applies after deformation.
   411→    // When all radii are 0 this is equivalent to the straight-edged polygon.
   412→    const radii = resolveRectBorderRadius(props.borderRadius);
   413→    const pathD = buildRoundedPolygonPath(corners, radii, { x: element.x, y: element.y });
   414→    const svgIdPrefix = `prevrect-${element.id}`;
   415→    const fillDesc = buildSvgFillDescriptor(effects.gradient, svgIdPrefix, props.fill);
   416→    const filterCSS = buildSvgFilterCSS(effects);
   417→    const dasharray = svgStrokeDasharray(props.borderStyle, props.borderWidth);
   418→    const borderStyleValue =
   419→      props.borderStyle === 'dotted' ? 'dotted' : props.borderStyle === 'dashed' ? 'dashed' : 'solid';
   420→    const reflectionMask = effects.reflection.enabled ? getReflectionMask(effects.reflection.preset) : undefined;
   421→    return (
   422→      <div style={{ position: 'relative', width: '100%', height: '100%' }}>
   423→        <svg
   424→          width="100%"
   425→          height="100%"
   426→          viewBox={`0 0 ${element.width} ${element.height}`}
   427→          overflow="visible"
   428→          style={{ position: 'relative', zIndex: 2, display: 'block', filter: filterCSS || undefined, opacity: props.opacity }}
   429→        >
   430→          {fillDesc.defs ? <g dangerouslySetInnerHTML={{ __html: fillDesc.defs }} /> : null}
   431→          <path
   432→            d={pathD}
   433→            fill={fillDesc.fill}
   434→            stroke={props.borderColor}
   435→            strokeWidth={props.borderWidth}
   436→            strokeDasharray={dasharray}
   437→            strokeLinejoin="round"
   438→            strokeLinecap={borderStyleValue === 'dotted' ? 'round' : 'butt'}
   439→          />
   440→        </svg>
   441→        {effects.reflection.enabled && (
   442→          <div
   443→            style={{
   444→              position: 'absolute',
   445→              top: `calc(100% + ${effects.reflection.distance}px)`,
   446→              left: 0,
   447→              width: '100%',
   448→              height: `${effects.reflection.size}%`,
   449→              overflow: 'hidden',
   450→              zIndex: 1,
   451→              pointerEvents: 'none',
   452→              maskImage: reflectionMask,
   453→              WebkitMaskImage: reflectionMask,
   454→            }}
   455→          >
   456→            <svg
   457→              width="100%"
   458→              height="100%"
   459→              viewBox={`0 0 ${element.width} ${element.height}`}
   460→              overflow="visible"
   461→              style={{ display: 'block', transform: 'scaleY(-1)', opacity: effects.reflection.opacity }}
   462→            >
   463→              {fillDesc.defs ? <g dangerouslySetInnerHTML={{ __html: fillDesc.defs }} /> : null}
   464→              <path
   465→                d={pathD}
   466→                fill={fillDesc.fill}
   467→                stroke={props.borderColor}
   468→                strokeWidth={props.borderWidth}
   469→                strokeDasharray={dasharray}
   470→                strokeLinejoin="round"
   471→              />
   472→            </svg>
   473→          </div>
   474→        )}
   475→      </div>
   476→    );
   477→  }
   478→
   479→  // Build box-shadow combining glow + shadow
   480→  const glowCSS = buildGlowCSS(effects.glow);
   481→  const shadowCSS = buildShadowCSS(effects.shadow, typeof props.borderRadius === 'number' ? props.borderRadius : resolveRectBorderRadius(props.borderRadius).topLeft);
   482→  let boxShadow = '';
   483→  if (effects.glow.enabled && effects.shadow.enabled) {
   484→    const glowShadow = glowCSS.boxShadow || '';
   485→    const dropShadow = shadowCSS.boxShadow || '';
   486→    boxShadow = `${dropShadow}, ${glowShadow}`;
   487→  } else if (effects.glow.enabled) {
   488→    boxShadow = (glowCSS.boxShadow as string) || '';
   489→  } else if (effects.shadow.enabled) {
   490→    boxShadow = (shadowCSS.boxShadow as string) || '';
   491→  }
   492→
   493→  // Build background using only longhand CSS properties to avoid
   494→  // React warnings about mixing shorthand/longhand for same value.
   495→  const bgStyle: React.CSSProperties = effects.gradient.type === 'transparent'
   496→    ? {} // Transparent — no background
   497→    : effects.gradient.type === 'gradient'
   498→      ? { backgroundImage: buildGradientCSS(effects.gradient, element.width, element.height) }
   499→      : effects.gradient.type === 'picture'
   500→        ? { backgroundColor: props.fill, ...buildPictureFillCSS(effects.gradient) }
   501→        : { backgroundColor: props.fill };
   502→
   503→  return (
   504→    <div style={{ position: 'relative', width: '100%', height: '100%' }}>
   505→      <div style={{
   506→        width: '100%',
   507→        height: '100%',
   508→        ...bgStyle,
   509→        borderWidth: `${props.borderWidth}px`,
   510→        borderColor: props.borderColor,
   511→        borderStyle: props.borderStyle === 'dotted' ? 'dotted' : props.borderStyle === 'dashed' ? 'dashed' : 'solid',
   512→        borderRadius: (() => { const r = resolveRectBorderRadius(props.borderRadius); return `${r.topLeft}px ${r.topRight}px ${r.bottomRight}px ${r.bottomLeft}px`; })(),
   513→        opacity: props.opacity,
   514→        boxSizing: 'border-box',
   515→        ...(boxShadow ? { boxShadow } : {}),
   516→        position: 'relative',
   517→        zIndex: 2,
   518→      }} />
   519→      {effects.reflection.enabled && (() => {
   520→        const maskGradient = REFLECTION_MASKS[effects.reflection.preset] || REFLECTION_MASKS.fade;
   521→        const radii = resolveRectBorderRadius(props.borderRadius);
   522→        const borderRadiusStr = `${radii.topLeft}px ${radii.topRight}px ${radii.bottomRight}px ${radii.bottomLeft}px`;
   523→        return (
   524→          <div style={{
   525→            position: 'absolute',
   526→            top: `calc(100% + ${effects.reflection.distance}px)`,
   527→            left: 0,
   528→            width: '100%',
   529→            height: `${effects.reflection.size}%`,
   530→            overflow: 'hidden',
   531→            zIndex: 1,
   532→            pointerEvents: 'none',
   533→            WebkitMaskImage: maskGradient,
   534→            maskImage: maskGradient,
   535→          }}>
   536→            <div style={{
   537→              width: '100%',
   538→              height: '100%',

   555→function PreviewEllipseElement({ element }: { element: CanvasElement }) {
   556→  const props = (element.properties as { type: 'ellipse'; data: EllipseProperties }).data;
   557→  const effects: RectangleEffects = props.effects ?? { ...DEFAULT_EFFECTS };
   558→
   559→  // Build box-shadow combining glow + shadow
   560→  const glowCSS = buildGlowCSS(effects.glow);
   561→  const shadowCSS = buildShadowCSS(effects.shadow, 50);
   562→  let boxShadow = '';
   563→  if (effects.glow.enabled && effects.shadow.enabled) {
   564→    const glowShadow = glowCSS.boxShadow || '';
   565→    const dropShadow = shadowCSS.boxShadow || '';
   566→    boxShadow = `${dropShadow}, ${glowShadow}`;
   567→  } else if (effects.glow.enabled) {
   568→    boxShadow = (glowCSS.boxShadow as string) || '';
   569→  } else if (effects.shadow.enabled) {
   570→    boxShadow = (shadowCSS.boxShadow as string) || '';
   571→  }
   572→
   573→  // Build background using only longhand CSS properties to avoid
   574→  // React warnings about mixing shorthand/longhand for same value.
   575→  const bgStyle: React.CSSProperties = effects.gradient.type === 'transparent'
   576→    ? {} // Transparent — no background
   577→    : effects.gradient.type === 'gradient'
   578→      ? { backgroundImage: buildGradientCSS(effects.gradient, element.width, element.height) }
   579→      : effects.gradient.type === 'picture'
   580→        ? { backgroundColor: props.fill, ...buildPictureFillCSS(effects.gradient) }
   581→        : { backgroundColor: props.fill };
   582→
   583→  return (
   584→    <div style={{ position: 'relative', width: '100%', height: '100%' }}>
   585→      <div style={{
   586→        width: '100%',
   587→        height: '100%',
   588→        ...bgStyle,
   589→        borderWidth: `${props.borderWidth}px`,
   590→        borderColor: props.borderColor,
   591→        borderStyle: props.borderStyle === 'dotted' ? 'dotted' : props.borderStyle === 'dashed' ? 'dashed' : 'solid',
   592→        borderRadius: '50%',
   593→        opacity: props.opacity,
   594→        boxSizing: 'border-box',
   595→        ...(boxShadow ? { boxShadow } : {}),
   596→        position: 'relative',
   597→        zIndex: 2,
   598→      }} />
   599→      {effects.reflection.enabled && (() => {
   600→        const maskGradient = REFLECTION_MASKS[effects.reflection.preset] || REFLECTION_MASKS.fade;
   601→        return (
   602→          <div style={{
   603→            position: 'absolute',
   604→            top: `calc(100% + ${effects.reflection.distance}px)`,
   605→            left: 0,
   606→            width: '100%',
   607→            height: `${effects.reflection.size}%`,
   608→            overflow: 'hidden',
   609→            zIndex: 1,
   610→            pointerEvents: 'none',
   611→            WebkitMaskImage: maskGradient,
   612→            maskImage: maskGradient,
   613→          }}>
   614→            <div style={{
   615→              width: '100%',
   616→              height: '100%',
   617→              transform: 'scaleY(-1)',
   618→              opacity: effects.reflection.opacity,
   619→              ...bgStyle,
   620→              borderWidth: `${props.borderWidth}px`,
   621→              borderColor: props.borderColor,
   622→              borderStyle: props.borderStyle === 'dotted' ? 'dotted' : props.borderStyle === 'dashed' ? 'dashed' : 'solid',
   623→              borderRadius: '50%',
   624→              boxSizing: 'border-box',
   625→            }} />
   626→          </div>
   627→        );
   628→      })()}
   629→    </div>
   630→  );

   633→// Renders the visual content of a child element inside a group (non-interactive)
   634→function PreviewGroupChildContent({ element }: { element: CanvasElement }) {
   635→  if (!element.visible) return null;

   642→    case 'ellipse': return <PreviewEllipseElement element={element} />;

   674→const PreviewElement = memo(function PreviewElement({ element }: { element: CanvasElement }) {
   675→  if (!element.visible) return null;
   676→
   677→  // For rectangles with rotation, wrap the shape content in a rotating div
   678→  // (same "rotate content only" strategy as the canvas + PDF renderers — the
   679→  // outer positioning wrapper stays axis-aligned so the absolute left/top/
   680→  // width/height = bbox is correct; only the inner shape spins).
   681→  const rectRotation = element.type === 'rectangle' ? getRectRotation(element) : 0;
   682→  const rotationWrapperStyle = rectRotation
   683→    ? { transform: `rotate(${rectRotation}deg)`, transformOrigin: 'center' as const, width: '100%' as const, height: '100%' as const }
   684→    : { width: '100%' as const, height: '100%' as const };
   685→
   686→  return (
   687→    <div
   688→      style={{
   689→        position: 'absolute',
   690→        left: element.x,
   691→        top: element.y,
   692→        width: element.width,
   693→        height: element.height,
   694→      }}
   695→    >
   696→      <div style={rotationWrapperStyle}>
   697→        {element.type === 'text' && <PreviewTextElement element={element} />}
   698→        {element.type === 'table' && <PreviewTableElement element={element} />}
   699→        {element.type === 'image' && <PreviewImageElement element={element} />}
   700→        {element.type === 'line' && <PreviewLineElement element={element} />}
   701→        {element.type === 'rectangle' && <PreviewRectangleElement element={element} />}
   702→        {element.type === 'ellipse' && <PreviewEllipseElement element={element} />}
   703→        {element.type === 'group' && <PreviewGroupElement element={element} />}
   704→      </div>
   705→    </div>
   706→  );
   707→});

   711→export function LargePreviewModal() {
   712→  const largePreviewOpen = useDesignerStore((s) => s.largePreviewOpen);
   713→  const setLargePreviewOpen = useDesignerStore((s) => s.setLargePreviewOpen);
   714→  const [zoom, setZoom] = useState(1);
   715→  const [zoomStep, setZoomStep] = useState(15);
   716→  const [customStep, setCustomStep] = useState('');
   717→  const [showCustomStep, setShowCustomStep] = useState(false);
   718→  // Dark preview background — toggles the preview workspace area to a dark
   719→  // grey so the user can preview how the invoice looks on a dark backdrop.
   720→  // Only affects the area around the page, not the page itself.
   721→  const [darkPreview, setDarkPreview] = useState(false);
   722→  const containerRef = useRef<HTMLDivElement>(null);
   723→  const zoomRef = useRef(1);
   724→  // Ref to queue a scroll-adjustment that must happen after the next render
   725→  const pendingScrollAdjust = useRef<{ contentX: number; contentY: number } | null>(null);
   726→
   727→  const pageWidth = useDesignerStore((s) => s.canvasSettings.pageWidth);
   728→  const pageHeight = useDesignerStore((s) => s.canvasSettings.pageHeight);
   729→  const pageBackgroundColor = useDesignerStore((s) => s.canvasSettings.pageBackgroundColor);
   730→  const elements = useDesignerStore((s) => s.elements);
   731→
   732→  // Apply pending scroll adjustments synchronously before the browser paints
   733→  // so the user never sees a frame with wrong scroll position
   734→  useLayoutEffect(() => {
   735→    const adj = pendingScrollAdjust.current;
   736→    if (adj === null) return;
   737→    pendingScrollAdjust.current = null;
   738→
   739→    const container = containerRef.current;
   740→    if (!container) return;
   741→
   742→    const centerX = container.clientWidth / 2;
   743→    const centerY = container.clientHeight / 2;
   744→    container.scrollLeft = adj.contentX * zoom - centerX;
   745→    container.scrollTop = adj.contentY * zoom - centerY;
   746→  }, [zoom]);
   747→
   748→  // Close on Escape
   749→  useEffect(() => {
   750→    if (!largePreviewOpen) return;
   751→    const handleKeyDown = (e: KeyboardEvent) => {
   752→      if (e.key === 'Escape') {
   753→        setLargePreviewOpen(false);
   754→      }
   755→    };
   756→    window.addEventListener('keydown', handleKeyDown);
   757→    return () => window.removeEventListener('keydown', handleKeyDown);
   758→  }, [largePreviewOpen, setLargePreviewOpen]);
   759→
   760→  // Mouse wheel zoom — RAF-throttled to coalesce rapid scroll events
   761→  const wheelAccumRef = useRef({ delta: 0, contentX: 0, contentY: 0 });
   762→  const wheelRafRef = useRef<number | null>(null);
   763→
   764→  useEffect(() => {
   765→    if (!largePreviewOpen) return;
   766→    const container = containerRef.current;
   767→    if (!container) return;
   768→
   769→    const handleWheel = (e: WheelEvent) => {
   770→      e.preventDefault();
   771→
   772→      const currentZoom = zoomRef.current;
   773→      const step = (e.ctrlKey || e.metaKey ? 5 : zoomStep) / 100;
   774→      const delta = e.deltaY > 0 ? -step : step;
   775→
   776→      // Compute content point at viewport center BEFORE zoom changes
   777→      const centerX = container.clientWidth / 2;
   778→      const centerY = container.clientHeight / 2;
   779→      const contentX = (container.scrollLeft + centerX) / currentZoom;
   780→      const contentY = (container.scrollTop + centerY) / currentZoom;
   781→
   782→      // Accumulate and coalesce via RAF
   783→      wheelAccumRef.current.delta += delta;
   784→      wheelAccumRef.current.contentX = contentX;
   785→      wheelAccumRef.current.contentY = contentY;
   786→
   787→      if (wheelRafRef.current === null) {
   788→        wheelRafRef.current = requestAnimationFrame(() => {
   789→          const accum = wheelAccumRef.current;
   790→          const newZoom = Math.min(MAX_ZOOM, Math.max(MIN_ZOOM, currentZoom + accum.delta));
   791→          accum.delta = 0;
   792→
   793→          if (newZoom !== currentZoom) {
   794→            pendingScrollAdjust.current = { contentX: accum.contentX, contentY: accum.contentY };
   795→            zoomRef.current = newZoom;
   796→            setZoom(newZoom);
   797→          }
   798→
   799→          wheelRafRef.current = null;
   800→        });
   801→      }
   802→    };
   803→
   804→    container.addEventListener('wheel', handleWheel, { passive: false });
   805→    return () => {
   806→      container.removeEventListener('wheel', handleWheel);
   807→      if (wheelRafRef.current !== null) {
   808→        cancelAnimationFrame(wheelRafRef.current);
   809→        wheelRafRef.current = null;
   810→      }
   811→    };
   812→  }, [largePreviewOpen, zoomStep]);
   813→
   814→  // Helper: update zoom and re-center on the viewport center
   815→  const applyZoomCentered = useCallback((newZoom: number) => {
   816→    const container = containerRef.current;
   817→    if (!container) return;
   818→
   819→    const currentZoom = zoomRef.current;
   820→    const clamped = Math.min(MAX_ZOOM, Math.max(MIN_ZOOM, newZoom));
   821→
   822→    // Compute content point at viewport center BEFORE updating zoom
   823→    const centerX = container.scrollLeft + container.clientWidth / 2;
   824→    const centerY = container.scrollTop + container.clientHeight / 2;
   825→    const contentX = centerX / currentZoom;
   826→    const contentY = centerY / currentZoom;
   827→
   828→    // Queue scroll adjustment for useLayoutEffect (runs before paint)
   829→    pendingScrollAdjust.current = { contentX, contentY };
   830→    zoomRef.current = clamped;
   831→    setZoom(clamped);
   832→  }, []);
   833→
   834→  // Fit to view

   840→    const containerH = container.clientHeight - 48;
   841→    const fitZoom = Math.min(containerW / pageWidth, containerH / pageHeight, 1);
   842→
   843→    // Center the page in the viewport via pendingScrollAdjust
   844→    pendingScrollAdjust.current = { contentX: pageWidth / 2, contentY: pageHeight / 2 };
   845→    zoomRef.current = fitZoom;
   846→    setZoom(fitZoom);
   847→  }, [pageWidth, pageHeight]);
   848→
   849→  // Fit to view when modal opens
   850→  useEffect(() => {
   851→    if (!largePreviewOpen) return;
   852→    // Wait for the container to mount and layout before calculating fit zoom
   853→    const raf = requestAnimationFrame(() => {
   854→      handleFitToView();
   855→    });
   856→    return () => cancelAnimationFrame(raf);
   857→  }, [largePreviewOpen, handleFitToView]);
   858→

   862→    setShowCustomStep(false);
   863→    setCustomStep('');
   864→  };
   865→

   871→      setCustomStep('');
   872→    }
   873→  };
   874→

   877→

   879→  const visibleElements = useMemo(() => elements.filter((el) => el.visible), [elements]);

   884→  const scaledHeight = pageHeight * zoom;
   885→
