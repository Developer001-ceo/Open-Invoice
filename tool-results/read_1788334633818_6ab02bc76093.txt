    62→    62→// ── Re-ID element helper ────────────────────────────────────────────────────
    63→    63→// When duplicating/pasting an element (especially a group), we must assign new
    64→    64→// UUIDs to the element AND all of its children to avoid duplicate IDs in the
    65→    65→// element tree.  Duplicate IDs cause lookups (findElementById, findParentGroup)
    66→    66→// to return the wrong element.
    67→    67→function reIdElement(element: CanvasElement): CanvasElement {
    68→    68→  const result: CanvasElement = {
    69→    69→    ...element,
    70→    70→    id: uuidv4(),
    71→    71→    // Deep-copy cornerPoints so duplicates/pastes have independent corners
    72→    72→    // (a shallow copy would share the array reference with the source).
    73→    73→    ...(element.cornerPoints
    74→    74→      ? { cornerPoints: element.cornerPoints.map((p) => ({ x: p.x, y: p.y })) }
    75→    75→      : {}),
    76→    76→  };
    77→    77→  if (result.type === 'group' && result.children) {
    78→    78→    result.children = result.children.map((c) => reIdElement(c));
    79→    79→  }
    80→    80→  return result;
    81→    81→}

    83→    83→// ── Line–Rectangle intersection ────────────────────────────────────────────
    84→    84→// Used by marquee selection to accurately test whether a line element
    85→    85→// intersects the selection rectangle, rather than relying on the line's
    86→    86→// bounding box which can be much larger than the visible line itself.
    87→    87→
    88→    88→/** Check if point (px, py) lies inside the axis-aligned rect [l,t,r,b]. */
    89→    89→function pointInRect(px: number, py: number, l: number, t: number, r: number, b: number): boolean {
    90→    90→  return px >= l && px <= r && py >= t && py <= b;
    91→    91→}
    92→    92→
    93→    93→/** Return the cross product of vectors (p1→p2) × (p1→p3). */
    94→    94→function cross(p1x: number, p1y: number, p2x: number, p2y: number, p3x: number, p3y: number): number {
    95→    95→  return (p2x - p1x) * (p3y - p1y) - (p2y - p1y) * (p3x - p1x);
    96→    96→}
    97→    97→
    98→    98→/**
    99→    99→ * Check whether the line segment (x1,y1)→(x2,y2) intersects the
   100→   100→ * axis-aligned rectangle [left, top, right, bottom].
   101→   101→ *
   102→   102→ * Algorithm:
   103→   103→ *  1. If either endpoint is inside the rect → intersect.
   104→   104→ *  2. Test intersection of the segment against each of the 4 rectangle
   105→   105→ *     edges using the separating-axis (cross-product) test.
   106→   106→ */
   107→   107→function lineIntersectsRect(
   108→   108→  x1: number, y1: number, x2: number, y2: number,
   109→   109→  left: number, top: number, right: number, bottom: number,
   110→   110→): boolean {
   111→   111→  // Quick check: either endpoint inside the rect
   112→   112→  if (pointInRect(x1, y1, left, top, right, bottom)) return true;
   113→   113→  if (pointInRect(x2, y2, left, top, right, bottom)) return true;
   114→   114→
   115→   115→  // Check intersection with each of the 4 edges of the rectangle
   116→   116→  // Top edge: (left,top)→(right,top)
   117→   117→  if (segmentsIntersect(x1, y1, x2, y2, left, top, right, top)) return true;
   118→   118→  // Bottom edge: (left,bottom)→(right,bottom)
   119→   119→  if (segmentsIntersect(x1, y1, x2, y2, left, bottom, right, bottom)) return true;
   120→   120→  // Left edge: (left,top)→(left,bottom)
   121→   121→  if (segmentsIntersect(x1, y1, x2, y2, left, top, left, bottom)) return true;
   122→   122→  // Right edge: (right,top)→(right,bottom)
   123→   123→  if (segmentsIntersect(x1, y1, x2, y2, right, top, right, bottom)) return true;
   124→   124→
   125→   125→  return false;
   126→   126→}
   127→   127→
   128→   128→/**
   129→   129→ * Check whether two line segments (ax1,ay1)→(ax2,ay2) and
   130→   130→ * (bx1,by1)→(bx2,by2) properly intersect (crossing each other,
   131→   131→ * not just touching at endpoints).
   132→   132→ */
   133→   133→function segmentsIntersect(
   134→   134→  ax1: number, ay1: number, ax2: number, ay2: number,
   135→   135→  bx1: number, by1: number, bx2: number, by2: number,
   136→   136→): boolean {
   137→   137→  const d1 = cross(bx1, by1, bx2, by2, ax1, ay1);
   138→   138→  const d2 = cross(bx1, by1, bx2, by2, ax2, ay2);
   139→   139→  const d3 = cross(ax1, ay1, ax2, ay2, bx1, by1);
   140→   140→  const d4 = cross(ax1, ay1, ax2, ay2, bx2, by2);
   141→   141→
   142→   142→  if (((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) &&
   143→   143→      ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0))) {
   144→   144→    return true;
   145→   145→  }
   146→   146→
   147→   147→  // Collinear / endpoint-touching cases — treat as non-intersecting for
   148→   148→  // marquee selection purposes (the endpoint-inside-rect check above
   149→   149→  // already handles the case where a line endpoint is inside the rect).
   150→   150→  return false;
   151→   151→}

   154→   154→const MAX_UNDO_HISTORY = 50;
   155→   155→
   156→   156→/** Subset of store state that represents a "document" snapshot for undo/redo. */
   157→   157→interface HistorySnapshot {
   158→   158→  elements: CanvasElement[];
   159→   159→  elementCounter: Record<ElementType, number>;
   160→   160→  canvasSettings: CanvasSettings;

   163→   163→function takeSnapshot(state: {
   164→   164→  elements: CanvasElement[];
   165→   165→  elementCounter: Record<ElementType, number>;
   166→   166→  canvasSettings: CanvasSettings;
   167→   167→}): HistorySnapshot {
   168→   168→  return {
   169→   169→    elements: structuredClone(state.elements),
   170→   170→    elementCounter: { ...state.elementCounter },
   171→   171→    canvasSettings: structuredClone(state.canvasSettings),
   172→   172→  };

   175→   175→// Debounce state for property update undo pushes.
   176→   176→// Prevents flooding the undo stack when sliders or text inputs fire many rapid changes.
   177→   177→let propertyUndoTimer: ReturnType<typeof setTimeout> | null = null;
   178→   178→let pendingPropertySnapshot: HistorySnapshot | null = null;
   179→   179→const PROPERTY_UNDO_DEBOUNCE_MS = 400;
   180→   180→
   181→   181→/** Internal helper: push a snapshot onto the undo stack and clear redo. */
   182→   182→function commitSnapshotToUndo(snapshot: HistorySnapshot) {
   183→   183→  const current = useDesignerStore.getState();
   184→   184→  const newUndoStack = [...current.undoStack, snapshot];
   185→   185→  if (newUndoStack.length > MAX_UNDO_HISTORY) newUndoStack.shift();
   186→   186→  useDesignerStore.setState({
   187→   187→    undoStack: newUndoStack,
   188→   188→    redoStack: [],
   189→   189→    canUndo: true,
   190→   190→    canRedo: false,
   191→   191→  });
   192→   192→}
   193→   193→
   194→   194→/** Push a debounced undo snapshot for property changes.
   195→   195→ *  The first call captures the "before" state, immediately clears the redo
   196→   196→ *  stack and updates canUndo/canRedo so the UI stays consistent. Subsequent
   197→   197→ *  calls within the debounce window are absorbed. When the timer fires, the
   198→   198→ *  snapshot is pushed onto the undo stack. */
   199→   199→function debouncedPropertyPushUndo(
   200→   200→  state: { elements: CanvasElement[]; elementCounter: Record<ElementType, number>; canvasSettings: CanvasSettings },
   201→   201→) {
   202→   202→  if (propertyUndoTimer === null) {
   203→   203→    // First call in a new interaction — capture the BEFORE state immediately
   204→   204→    pendingPropertySnapshot = takeSnapshot(state);
   205→   205→    // Immediately clear redo stack & update flags so the UI is consistent
   206→   206→    // from the moment the first property change happens.
   207→   207→    useDesignerStore.setState({
   208→   208→      redoStack: [],
   209→   209→      canRedo: false,
   210→   210→      canUndo: true, // the pending snapshot guarantees at least one undo step
   211→   211→    });
   212→   212→  }
   213→   213→  // Reset the debounce timer
   214→   214→  if (propertyUndoTimer) clearTimeout(propertyUndoTimer);
   215→   215→  propertyUndoTimer = setTimeout(() => {
   216→   216→    if (pendingPropertySnapshot !== null) {
   217→   217→      // Push the captured "before" snapshot
   218→   218→      commitSnapshotToUndo(pendingPropertySnapshot);
   219→   219→      pendingPropertySnapshot = null;
   220→   220→    }
   221→   221→    propertyUndoTimer = null;
   222→   222→  }, PROPERTY_UNDO_DEBOUNCE_MS);
   223→   223→}
   224→   224→
   225→   225→/** Immediately flush any pending debounced undo entry (call before non-debounced actions).
   226→   226→ *  Returns true if a pending snapshot was committed. */
   227→   227→function flushPropertyUndo(): boolean {
   228→   228→  if (propertyUndoTimer !== null) {
   229→   229→    clearTimeout(propertyUndoTimer);
   230→   230→    if (pendingPropertySnapshot !== null) {
   231→   231→      commitSnapshotToUndo(pendingPropertySnapshot);
   232→   232→      pendingPropertySnapshot = null;
   233→   233→    }
   234→   234→    propertyUndoTimer = null;
   235→   235→    return true;
   236→   236→  }
   237→   237→  return false;
   238→   238→}
   239→   239→
   240→   240→/** Cancel any pending debounced undo entry WITHOUT committing it.
   241→   241→ *  Used when undo/redo is invoked — the pending "before" state is already
   242→   242→ *  what the user wants to go back to, so we don't need to push it. */
   243→   243→function cancelPropertyUndo() {
   244→   244→  if (propertyUndoTimer !== null) {
   245→   245→    clearTimeout(propertyUndoTimer);
   246→   246→    pendingPropertySnapshot = null;
   247→   247→    propertyUndoTimer = null;
   248→   248→  }
   249→   249→}
   250→   250→
   251→   251→/**
   252→   252→ * Pure recursive helper: returns a NEW elements tree with the properties of the
   253→   253→ * element identified by `id` replaced. Searches both top-level elements and the
   254→   254→ * `children` arrays of groups (arbitrarily deep, matching `findElementById`).
   255→   255→ *
   256→   256→ * Unlike the `updateElementProperties` store action, this is side-effect-free —
   257→   257→ * no undo snapshot is captured — so it is safe to call on every mousemove of a
   258→   258→ * drag whose undo entry was already pushed at drag start (e.g. table resize).
   259→   259→ */
   260→   260→function setElementPropertiesInTree(
   261→   261→  elements: CanvasElement[],
   262→   262→  id: string,
   263→   263→  properties: ElementProperties,
   264→   264→): CanvasElement[] {
   265→   265→  const updateChildren = (children: CanvasElement[]): CanvasElement[] => {
   266→   266→    const idx = children.findIndex((c) => c.id === id);
   267→   267→    if (idx !== -1) {
   268→   268→      const next = [...children];
   269→   269→      next[idx] = { ...next[idx], properties };
   270→   270→      return next;
   271→   271→    }
   272→   272→    // Recurse into nested groups (groups are normally single-level here, but
   273→   273→    // recurse anyway to stay consistent with findElementById).
   274→   274→    let changed = false;
   275→   275→    const maybeDeeper = children.map((c) => {
   276→   276→      if (c.type === 'group' && c.children) {
   277→   277→        const newGrand = updateChildren(c.children);
   278→   278→        if (newGrand !== c.children) {
   279→   279→          changed = true;
   280→   280→          return { ...c, children: newGrand };
   281→   281→        }
   282→   282→      }
   283→   283→      return c;
   284→   284→    });
   285→   285→    return changed ? maybeDeeper : children;
   286→   286→  };
   287→   287→
   288→   288→  let changed = false;
   289→   289→  const result = elements.map((el) => {
   290→   290→    if (el.id === id) {
   291→   291→      changed = true;
   292→   292→      return { ...el, properties };
   293→   293→    }
   294→   294→    if (el.type === 'group' && el.children) {
   295→   295→      const newChildren = updateChildren(el.children);
   296→   296→      if (newChildren !== el.children) {
   297→   297→        changed = true;
   298→   298→        return { ...el, children: newChildren };
   299→   299→      }
   300→   300→    }
   301→   301→    return el;
   302→   302→  });
   303→   303→  return changed ? result : elements;
   304→   304→}
