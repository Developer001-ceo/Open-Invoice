   199→    },
   200→  };
   201→
   202→  return (
   203→    <Dialog open={open} onOpenChange={onOpenChange}>
   204→      <DialogContent
   205→        className="max-w-[920px] w-[92vw] p-0 gap-0 overflow-hidden rounded-2xl border-border/60 shadow-[0_1px_2px_rgba(0,0,0,0.04),0_8px_24px_rgba(0,0,0,0.08),0_24px_48px_rgba(0,0,0,0.06)] ring-1 ring-inset ring-white/[0.06] dark:ring-white/[0.04]"
   206→        showCloseButton={false}
   207→        onInteractOutside={(e) => e.preventDefault()}
   208→        onEscapeKeyDown={(e) => e.preventDefault()}

   210→        {/* Film grain overlay — applies to entire modal */}
   211→        <div
   212→          className="pointer-events-none absolute inset-0 z-[60] opacity-[0.015] mix-blend-overlay"
   213→          aria-hidden="true"
   214→          style={{
   215→            backgroundImage:
   216→              "url(\"data:image/svg+xml,%3Csvg viewBox='0 0 200 200' xmlns='http://www.w3.org/2000/svg'%3E%3Cfilter id='n'%3E%3CfeTurbulence type='fractalNoise' baseFrequency='0.9' numOctaves='3' stitchTiles='stitch'/%3E%3C/filter%3E%3Crect width='100%25' height='100%25' filter='url(%23n)'/%3E%3C/svg%3E\")",
   217→            backgroundRepeat: 'repeat',
   218→            backgroundSize: '200px 200px',
   219→          }}
   220→        />
   221→
   222→        <motion.div
   223→          variants={containerVariants}
   224→          initial="hidden"
   225→          animate="show"
   226→          className="grid grid-cols-[40%_60%] min-h-[560px] max-h-[88vh] overflow-x-hidden"
   227→        >
   228→          {/* ===================== LEFT: Branding Hero ===================== */}
   229→          <div className="relative bg-gradient-to-br from-primary/8 via-primary/4 to-transparent overflow-hidden flex flex-col">
   230→            {/* Animated mesh gradient blobs */}
   231→            <motion.div
   232→              className="absolute -top-20 -left-20 w-72 h-72 rounded-full bg-primary/10 blur-3xl"
   233→              animate={{
   234→                x: [0, 30, 0],
   235→                y: [0, 20, 0],
   236→                scale: [1, 1.1, 1],
   237→              }}
   238→              transition={{ duration: 18, repeat: Infinity, ease: 'easeInOut' }}
   239→            />
   240→            <motion.div
   241→              className="absolute bottom-0 right-0 w-64 h-64 rounded-full bg-amber-400/[0.04] blur-3xl"

   243→                x: [0, -25, 0],
   244→                y: [0, -15, 0],
   245→                scale: [1, 1.15, 1],
   246→              }}
   247→              transition={{ duration: 22, repeat: Infinity, ease: 'easeInOut' }}

   252→                x: [0, 20, 0],

   254→              }}
   255→              transition={{ duration: 16, repeat: Infinity, ease: 'easeInOut' }}
   256→            />
   257→
   258→            {/* Content */}
   259→            <div className="relative z-10 flex flex-col h-full px-8 pt-10 pb-8 flex-1">
   260→              {/* Sparkle accent + brand */}
   261→              <motion.div variants={itemVariants} className="flex items-center gap-2 mb-1">
   262→                <Sparkles className="h-3.5 w-3.5 text-primary/60" />
   263→                <span className="text-[10px] font-semibold uppercase tracking-[0.2em] text-muted-foreground/80">
   264→                  Invoice Studio

   267→
   268→              <motion.div variants={itemVariants} className="mt-2">
   269→                <h1

   271→                >
   272→                  Open Invoice
   273→                </h1>
   274→                {/* MJC brand mark — same size as the heading, gradient starts with primary on the left and ends with pink */}
   275→                <div className="mt-2 flex flex-col">
   276→                  <span className="text-3xl font-black tracking-[0.22em] leading-none bg-gradient-to-r from-primary via-rose-400 to-pink-500 bg-clip-text text-transparent">
   277→                    MJC
   278→                  </span>
   279→                  {/* Brand-signature flourish — thin gradient underline */}

   281→                </div>

   286→              </motion.div>
   287→
   288→              {/* Spacer */}

   290→
   291→              {/* Live canvas preview — actual size scaled to fit, with turtle logo watermark */}

   293→                <div className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/60 mb-2">
   294→                  Preview

   296→                <div
   297→                  className="relative flex items-center justify-center"
   298→                  style={{ height: PREVIEW_AREA_H }}
   299→                >
   300→                  {/* Page — uses actual pixel dimensions × scale, so different presets visibly differ */}

   323→                      style={{

   339→                      }}
   340→                    />
   341→                  </div>
   342→                </div>

   344→                  <span>{widthPx} × {heightPx}</span>
   345→                  <span className="text-muted-foreground/40">px</span>
   346→                </div>

   349→              {/* Version badge + hints toggle — bottom row of the branding hero.

   354→                <span className="text-[10px] text-muted-foreground/50">v1.0</span>

   366→                    className="scale-90 ml-0.5"
   367→                  />
   368→                </label>

   370→            </div>
   371→          </div>
   372→
   373→          {/* ===================== RIGHT: Form ===================== */}
   374→          <div ref={setRightColRef} className="relative bg-background overflow-y-auto overflow-x-hidden flex flex-col">
   375→            {/* Form content */}
   376→            <div className="relative flex flex-col h-full">
   377→              {/* Header */}
   378→              <motion.div variants={itemVariants} className="px-7 pt-7 pb-5 border-b border-border/50">
   379→                <DialogHeader className="space-y-0">
   380→                  <DialogTitle className="text-lg font-semibold tracking-tight">
   381→                    Create a new template

   383→                  <DialogDescription className="sr-only">
   384→                    Configure your invoice template settings

   386→                </DialogHeader>

   389→              {/* Scrollable form */}

   411→                    Canvas Size
