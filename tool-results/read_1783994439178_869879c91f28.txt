    13→# 清理函数：优雅关闭所有服务
    14→cleanup() {
    15→    echo ""
    16→    echo "🛑 正在关闭所有服务..."
    17→    
    18→    # 发送 SIGTERM 信号给所有子进程
    19→    for pid in $pids; do
    20→        if kill -0 "$pid" 2>/dev/null; then
    21→            service_name=$(ps -p "$pid" -o comm= 2>/dev/null || echo "unknown")
    22→            echo "   关闭进程 $pid ($service_name)..."
    23→            kill -TERM "$pid" 2>/dev/null
    24→        fi
    25→    done
    26→    
    27→    # 等待所有进程退出（最多等待 5 秒）
    28→    sleep 1
    29→    for pid in $pids; do
    30→        if kill -0 "$pid" 2>/dev/null; then
    31→            # 如果还在运行，等待最多 4 秒
    32→            timeout=4
    33→            while [ $timeout -gt 0 ] && kill -0 "$pid" 2>/dev/null; do
    34→                sleep 1
    35→                timeout=$((timeout - 1))
    36→            done
    37→            # 如果仍然在运行，强制关闭
    38→            if kill -0 "$pid" 2>/dev/null; then
    39→                echo "   强制关闭进程 $pid..."
    40→                kill -KILL "$pid" 2>/dev/null
    41→            fi

   138→===== .zscripts/.zscripts/supervise-server.sh =====
   139→#!/bin/bash
   140→# Self-restarting supervisor for the Next.js production server.
   141→# Restarts the server if it crashes, with a small backoff.
   142→cd /home/z/my-project
   143→export NODE_ENV=production
   144→export PORT=3000
   145→export HOSTNAME=0.0.0.0
   146→
   147→while true; do
   148→  echo "[$(date '+%H:%M:%S')] Starting next start..."
   149→  ./node_modules/.bin/next start -p 3000
   150→  EXIT_CODE=$?
   151→  echo "[$(date '+%H:%M:%S')] Server exited with code $EXIT_CODE. Restarting in 3s..."
   152→  sleep 3
   153→done

   155→===== .zscripts/dev.sh =====
   156→#!/bin/bash
   157→
   158→set -euo pipefail
   159→
   160→# 获取脚本所在目录（.zscripts）
   161→# 使用 $0 获取脚本路径（与 build.sh 保持一致）
   162→SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
   163→PROJECT_DIR="$(cd "$SCRIPT_DIR/.." && pwd)"
   164→
   165→log_step_start() {
   166→        local step_name="$1"
   167→        echo "=========================================="
   168→        echo "[$(date '+%Y-%m-%d %H:%M:%S')] Starting: $step_name"
   169→        echo "=========================================="
   170→        export STEP_START_TIME
   171→        STEP_START_TIME=$(date +%s)
   172→}
   173→
   174→log_step_end() {
   175→        local step_name="${1:-Unknown step}"
   176→        local end_time
   177→        end_time=$(date +%s)
   178→        local duration=$((end_time - STEP_START_TIME))
   179→        echo "=========================================="
   180→        echo "[$(date '+%Y-%m-%d %H:%M:%S')] Completed: $step_name"
   181→        echo "[LOG] Step: $step_name | Duration: ${duration}s"
   182→        echo "=========================================="
   183→        echo ""
   184→}
   185→
   186→start_mini_services() {
   187→        local mini_services_dir="$PROJECT_DIR/mini-services"
   188→        local started_count=0
   189→
   190→        log_step_start "Starting mini-services"
   191→        if [ ! -d "$mini_services_dir" ]; then
   192→                echo "Mini-services directory not found, skipping..."
   193→                log_step_end "Starting mini-services"
   194→                return 0
   195→        fi
   196→
   197→        echo "Found mini-services directory, scanning for sub-services..."
   198→
   199→        for service_dir in "$mini_services_dir"/*; do
   200→                if [ ! -d "$service_dir" ]; then
   201→                        continue
   202→                fi
   203→
   204→                local service_name
   205→                service_name=$(basename "$service_dir")
   206→                echo "Checking service: $service_name"
   207→
   208→                if [ ! -f "$service_dir/package.json" ]; then
   209→                        echo "[$service_name] No package.json found, skipping..."
   210→                        continue
   211→                fi
   212→
   213→                if ! grep -q '"dev"' "$service_dir/package.json"; then
   214→                        echo "[$service_name] No dev script found, skipping..."
   215→                        continue
   216→                fi
   217→
   218→                echo "Starting $service_name in background..."
   219→                (
   220→                        cd "$service_dir"
   221→                        echo "[$service_name] Installing dependencies..."
   222→                        bun install
   223→                        echo "[$service_name] Running bun run dev..."
   224→                        exec bun run dev
   225→                ) >"$PROJECT_DIR/.zscripts/mini-service-${service_name}.log" 2>&1 &
   226→
   227→                local service_pid=$!
   228→                echo "[$service_name] Started in background (PID: $service_pid)"
   229→                echo "[$service_name] Log: $PROJECT_DIR/.zscripts/mini-service-${service_name}.log"
   230→                disown "$service_pid" 2>/dev/null || true
   231→                started_count=$((started_count + 1))
   232→        done
   233→
   234→        echo "Mini-services startup completed. Started $started_count service(s)."
   235→        log_step_end "Starting mini-services"
   236→}
   237→
   238→wait_for_service() {
   239→        local host="$1"
   240→        local port="$2"
   241→        local service_name="$3"
   242→        local max_attempts="${4:-60}"
   243→        local attempt=1
   244→
   245→        echo "Waiting for $service_name to be ready on $host:$port..."
   246→
   247→        while [ "$attempt" -le "$max_attempts" ]; do
   248→                if curl -s --connect-timeout 2 --max-time 5 "http://$host:$port" >/dev/null 2>&1; then
   249→                        echo "$service_name is ready!"
   250→                        return 0
   251→                fi
   252→
   253→                echo "Attempt $attempt/$max_attempts: $service_name not ready yet, waiting..."
   254→                sleep 1
   255→                attempt=$((attempt + 1))
   256→        done
   257→
   258→        echo "ERROR: $service_name failed to start within $max_attempts seconds"
   259→        return 1
   260→}
   261→
   262→cleanup() {
   263→        if [ -n "${DEV_PID:-}" ] && kill -0 "$DEV_PID" >/dev/null 2>&1; then
   264→                echo "Stopping Next.js dev server (PID: $DEV_PID)..."
   265→                kill "$DEV_PID" >/dev/null 2>&1 || true
   266→        fi
   267→}
   268→
   269→trap cleanup EXIT INT TERM
   270→
   271→cd "$PROJECT_DIR"
   272→
   273→if ! command -v bun >/dev/null 2>&1; then
   274→        echo "ERROR: bun is not installed or not in PATH"
   275→        exit 1
   276→fi
   277→
   278→log_step_start "bun install"
   279→echo "[BUN] Installing dependencies..."
   280→bun install
   281→log_step_end "bun install"
   282→
   283→log_step_start "bun run db:push"
   284→echo "[BUN] Setting up database..."
   285→bun run db:push
   286→log_step_end "bun run db:push"
   287→
   288→log_step_start "Starting Next.js dev server"
   289→echo "[BUN] Starting development server..."
   290→bun run dev &
   291→DEV_PID=$!
   292→log_step_end "Starting Next.js dev server"
   293→
   294→log_step_start "Waiting for Next.js dev server"
   295→wait_for_service "localhost" "3000" "Next.js dev server"
   296→log_step_end "Waiting for Next.js dev server"
   297→
   298→log_step_start "Health check"
   299→echo "[BUN] Performing health check..."
   300→curl -fsS localhost:3000 >/dev/null
   301→echo "[BUN] Health check passed"
   302→log_step_end "Health check"
   303→
   304→start_mini_services
   305→
   306→echo "Next.js dev server is running in background (PID: $DEV_PID)."
   307→echo "Use 'kill $DEV_PID' to stop it."
   308→disown "$DEV_PID" 2>/dev/null || true
   309→unset DEV_PID

   311→===== .zscripts/mini-services-start.sh =====

   347→                kill -KILL "$pid" 2>/dev/null

   349→        fi

   351→    
   352→    echo "✅ 所有服务已关闭"

   354→
   355→main() {
   356→    echo "🚀 开始启动所有 mini services..."
   357→    
   358→    # 检查 dist 目录是否存在
   359→    if [ ! -d "$DIST_DIR" ]; then
   360→        echo "ℹ️  目录 $DIST_DIR 不存在"
   361→        return
   362→    fi
   363→    
   364→    # 查找所有 mini-service-*.js 文件
   365→    service_files=""
   366→    for file in "$DIST_DIR"/mini-service-*.js; do
   367→        if [ -f "$file" ]; then
   368→            if [ -z "$service_files" ]; then
   369→                service_files="$file"
   370→            else
   371→                service_files="$service_files $file"
   372→            fi
   373→        fi
   374→    done
   375→    
   376→    # 计算服务文件数量
   377→    service_count=0
   378→    for file in $service_files; do
   379→        service_count=$((service_count + 1))
   380→    done
   381→    
   382→    if [ $service_count -eq 0 ]; then
   383→        echo "ℹ️  未找到任何 mini service 文件"
   384→        return
   385→    fi
   386→    
   387→    echo "📦 找到 $service_count 个服务，开始启动..."
   388→    echo ""
   389→    
   390→    # 启动每个服务
   391→    for file in $service_files; do
   392→        service_name=$(basename "$file" .js | sed 's/mini-service-//')
   393→        echo "▶️  启动服务: $service_name..."
   394→        
   395→        # 使用 bun 运行服务（后台运行）
   396→        bun "$file" &
   397→        pid=$!
   398→        if [ -z "$pids" ]; then
   399→            pids="$pid"
   400→        else
   401→            pids="$pids $pid"
   402→        fi
   403→        
   404→        # 等待一小段时间检查进程是否成功启动
   405→        sleep 0.5
   406→        if ! kill -0 "$pid" 2>/dev/null; then
   407→            echo "❌ $service_name 启动失败"
   408→            # 从字符串中移除失败的 PID
   409→            pids=$(echo "$pids" | sed "s/\b$pid\b//" | sed 's/  */ /g' | sed 's/^ *//' | sed 's/ *$//')
   410→        else
   411→            echo "✅ $service_name 已启动 (PID: $pid)"
   412→        fi
   413→    done
   414→    
   415→    # 计算运行中的服务数量
   416→    running_count=0
   417→    for pid in $pids; do
   418→        if kill -0 "$pid" 2>/dev/null; then
   419→            running_count=$((running_count + 1))
   420→        fi
   421→    done
   422→    
   423→    echo ""
   424→    echo "🎉 所有服务已启动！共 $running_count 个服务正在运行"
   425→    echo ""
   426→    echo "💡 按 Ctrl+C 停止所有服务"
   427→    echo ""
   428→    
   429→    # 等待所有后台进程
   430→    wait
   431→}

   565→# Boot timeline observability: lets us attribute FC 412 (port 81 health check
   566→# failed in 120s) to image-pull vs start.sh-step latency. /proc/uptime gives
   567→# seconds since container start, so subtracting from current time isolates the
   568→# FC platform overhead from anything start.sh itself does.
   569→BOOT_TIMELINE=/tmp/boot-timeline.log
   570→: > "$BOOT_TIMELINE"
   571→# All helpers below guard /proc/uptime with `|| echo ?` so a missing file
   572→# (non-Linux dev shells, broken procfs) never trips set -e and kills bootstrap.
   573→boot_uptime() { awk '{print $1}' /proc/uptime 2>/dev/null || echo "?"; }
   574→boot_ts()     { date -u +%FT%TZ; }
   575→echo "[BOOT] start.sh entered at $(boot_ts) uptime=$(boot_uptime)s" | tee -a "$BOOT_TIMELINE"
   576→
   577→# Dump the full timeline on any exit. FC's 412 error body only captures stdout
   578→# tail; without this, all early-step durations get truncated and we lose the
   579→# evidence we need to pick a real fix.
   580→dump_boot_timeline() {
   581→    local rc=$?
   582→    echo "[BOOT] exit rc=$rc at $(boot_ts) uptime=$(boot_uptime)s" >> "$BOOT_TIMELINE"
   583→    echo ""
   584→    echo "==== BOOT TIMELINE (full) ===="
   585→    cat "$BOOT_TIMELINE" 2>/dev/null || true
   586→    echo "==== END BOOT TIMELINE ===="
   587→}
   588→trap dump_boot_timeline EXIT
   589→
   590→# 添加耗时统计函数
   591→log_step_start() {
   592→    local step_name="$1"
   593→    echo "=========================================="
   594→    echo "[$(date '+%Y-%m-%d %H:%M:%S')] Starting: $step_name"
   595→    echo "=========================================="
   596→    echo "[BOOT] $(boot_ts) start step=\"$step_name\" uptime=$(boot_uptime)s" >> "$BOOT_TIMELINE"
   597→    export STEP_START_TIME=$(date +%s)
   598→}
   599→
   600→log_step_end() {
   601→    local step_name="$1"
   602→    if [ -z "$step_name" ]; then
   603→        step_name="Unknown step"
   604→    fi
   605→    local end_time=$(date +%s)
   606→    local duration=$((end_time - STEP_START_TIME))
   607→    echo "=========================================="
   608→    echo "[$(date '+%Y-%m-%d %H:%M:%S')] Completed: $step_name"
   609→    echo "[LOG] Step: $step_name | Duration: ${duration}s"
   610→    echo "=========================================="
   611→    echo ""
   612→    echo "[BOOT] $(boot_ts) end   step=\"$step_name\" duration=${duration}s uptime=$(boot_uptime)s" >> "$BOOT_TIMELINE"
   613→}

   779→# Write .gitignore BEFORE any git op so git can prune skills/ as it walks the
   780→# working tree. Without this, `git add .` lstats every file under
   781→# /home/z/my-project/skills/ (~10k+ files from the 58 freshly-extracted official
   782→# skill zips), blows past the 60s boot budget, and trips the EXIT trap — FC

   784→# Incident 2026-06-09: agent-claw-cb40fa5b boot-looped 36 times in 12.5h.
   785→# The same .gitignore also keeps runtime git_commit (app/util/git.py, git add -A)
   786→# from hashing the skill tree on every user commit.
   787→#
   788→# Scope is intentionally narrow: only skills/ and node_modules/. Both are already
   789→# explicitly excluded in the /git/ls-tree handler (mcp.py:600
   790→# `iglob("!node_modules").iglob("!skills")`), so adding them to .gitignore
   791→# costs nothing visible. Wider entries (upload/, download/, db/, *.log, .venv/,
   792→# __pycache__/, .next/, dist/, build/) used to live here too but `ripgrep --files`
   793→# honors .gitignore by default, which silently hid those dirs from /git/ls-tree —
   794→# the dir tree the agent uses to enumerate user-visible project content. Keep
   795→# this list minimal; do not add anything that the ls-tree handler doesn't
   796→# already exclude.
   797→#
   798→# Auto-heal for instances polluted by b407e6a's broad .gitignore: that commit
   799→# wrote `upload/` `download/` `db/` etc. into the file, then `git add . && git
   800→# commit` tracked the .gitignore, so the broad version is baked into repo.tar
   801→# and self-perpetuates across cold starts. Detect via the unique triple
   802→# upload/+download/+db/ (sandbox-internal dir names that would never coincide
   803→# in a user-authored .gitignore) and rewrite to the narrow version. Runs once
   804→# per affected instance; after the rewrite + the next runtime git_commit, the
   805→# narrow version lands in tar and the trigger no longer fires (convergent).
   806→needs_gitignore_write=false
   807→if [ ! -f /home/z/my-project/.gitignore ]; then
   808→  needs_gitignore_write=true
   809→elif grep -qE '^upload/$' /home/z/my-project/.gitignore \
   810→     && grep -qE '^download/$' /home/z/my-project/.gitignore \
   811→     && grep -qE '^db/$' /home/z/my-project/.gitignore; then
   812→  echo "[BOOT] auto-heal: detected broad .gitignore from b407e6a, rewriting to narrow"
   813→  needs_gitignore_write=true
   814→fi
   815→if [ "$needs_gitignore_write" = "true" ]; then
   816→  cat > /home/z/my-project/.gitignore <<'EOF'
   817→skills/
   818→node_modules/
   819→EOF
   820→  chown z:z /home/z/my-project/.gitignore

   826→  # *which* sub-step hangs. Each command is wrapped in `timeout` so a real

   852→wait_for_service() {
   853→    local host="$1"
   854→    local port="$2"
   855→    local service_name="$3"
   856→    local max_attempts=60
   857→    local attempt=1
   858→    
   859→    echo "Waiting for $service_name to be ready on $host:$port..."
   860→    
   861→    while [ $attempt -le $max_attempts ]; do
   862→        if curl -s --connect-timeout 2 --max-time 5 "http://$host:$port" > /dev/null 2>&1; then
   863→            echo "$service_name is ready!"
   864→            return 0
   865→        fi
   866→        
   867→        echo "Attempt $attempt/$max_attempts: $service_name not ready yet, waiting..."
   868→        sleep 1
   869→        attempt=$((attempt + 1))
   870→    done
   871→    
   872→    echo "ERROR: $service_name failed to start within $((max_attempts * 2)) seconds"
   873→    return 1
